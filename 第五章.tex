\chapter{回归分析}
\begin{theorem}[5.1.1]
若对任意 $x$，
\[
E(Y \mid X=x)=a+bx,
\tag{5.1.4}
\]
则
\[
E(Y \mid X=x)
=
\mu_2
+
\rho
\sqrt{\frac{\operatorname{Var}(Y)}{\operatorname{Var}(X)}}
(x-\mu_1),
\]
其中
\[
\mu_1=E(X),\qquad
\mu_2=E(Y),\qquad
\rho=
\frac{\operatorname{Cov}(X,Y)}
{\sqrt{\operatorname{Var}(X)\operatorname{Var}(Y)}}.
\]
\end{theorem}

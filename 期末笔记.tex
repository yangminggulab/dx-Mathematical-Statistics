\chapter{基础工具层}
\begin{note}[统计量是什么问题的答案]
你面对的基本困境是：总体分布未知（比如不知道 $\mu$ 是多少），手里只有 $X_1, X_2, \ldots, X_n$ 这 $n$ 个数。

$n$ 个数太多了，你不能直接用。你需要把 $n$ 个数压缩成一个数，用来代表"关于 $\theta$ 的信息"。这个压缩函数 $T(X_1,\ldots,X_n)$ 就是\textbf{统计量}。

唯一的限制：$T$ 里不能含 $\theta$ 本身——否则你等于用了还没求的答案，循环了。
\end{note}
\begin{definition}[常见统计量]
设 $X_1, X_2, \ldots, X_n$ 为样本，常用统计量有：

样本均值 $\displaystyle\bar{X} = \frac{1}{n}\sum_{i=1}^n X_i$，

样本方差 $\displaystyle s^2 = \frac{1}{n-1}\sum_{i=1}^n (X_i - \bar{X})^2$，

极差 $R = X_{(n)} - X_{(1)}$。
\end{definition}
\section{顺序统计量}

\begin{example}{为什么需要顺序统计量：从电路寿命说起}
你有 $n = 5$ 个灯泡，串联组成一条电路——\textbf{一个坏了，全线中断}。
问：这条电路能撑多久？

你知道每个灯泡的寿命 $X_i$ 服从同一分布 $F(x)$，也可以算出样本均值 $\bar{X}$。
但 $\bar{X}$ 在这里毫无用处：它告诉你的是"平均每个灯泡能用多久"，
而电路的寿命取决于\textbf{最先坏掉的那个}，即
\[
    T_{\text{串联}} = \min\{X_1, X_2, X_3, X_4, X_5\} = X_{(1)}.
\]
如果改成并联电路——\textbf{全部坏了才断}，则寿命取决于\textbf{最后一个}：
\[
    T_{\text{并联}} = \max\{X_1, X_2, X_3, X_4, X_5\} = X_{(n)}.
\]

\begin{itemize}
    \item $\bar{X}$ 描述数据的"整体水平"，把每个点的位置信息\textbf{平均掉了}。
    \item $X_{(1)}, \dots, X_{(n)}$ 保留了"第几名"的位置信息，
          专门回答极值、中间值、排名类问题。
\end{itemize}

凡是问题出现"最大""最小""第 $r$ 小""中位数"，本质上都是在问顺序统计量——
这就是它被发明的根本原因。
\end{example}
\begin{dxtips}
    均值方差是"融合"数据，顺序统计量是"保留位置"的数据。
    一类问题问平均水平用前者；一类问题问第几名用后者。
    两种工具解决两类完全不同的问题。
\end{dxtips}
\begin{theorem}[第 $r$ 个顺序统计量的密度]
设 $X_1, \ldots, X_n \overset{\text{i.i.d.}}{\sim} F(x)$，密度为 $f(x)$，则 $X_{(r)}$ 的密度为
$$f_{(r)}(x) = r C_n^r\, F^{r-1}(x)\,[1-F(x)]^{n-r}\,f(x)$$
\end{theorem}

\begin{note}[拆解记忆]
\begin{itemize}
  \item $f(x)$：第 $r$ 个恰好落在 $x$ 处
  \item $F^{r-1}(x)$：有 $r-1$ 个已经落在 $x$ 左边
  \item $[1-F(x)]^{n-r}$：有 $n-r$ 个还在 $x$ 右边
  \item $r C_n^r = \dfrac{n!}{(r-1)!(n-r)!}$：这三组人的全部排列方式数
\end{itemize}
\end{note}
\begin{note}[顺序统计量]
顺序统计量在一个特定情况下才出现：\textbf{参数出现在分布的支撑边界上（即参数决定数据的范围）。}

\textbf{用顺序统计量：}
\begin{itemize}
  \item $U(\theta_1, \theta_2)$：数据必须在 $[\theta_1, \theta_2]$ 里，参数就是边界 $\rightarrow$ 用 $X_{(1)}$、$X_{(n)}$
  \item $U(0, \theta)$：数据在 $[0, \theta]$ 里，$\theta$ 是上界 $\rightarrow$ 用 $X_{(n)}$
  \item 双参数指数分布 $f(x) = e^{-(x-\theta)}$，$x \geq \theta$：$\theta$ 是数据的下界 $\rightarrow$ 用 $X_{(1)}$
\end{itemize}

\textbf{不用顺序统计量：}
\begin{itemize}
  \item $N(\mu, \sigma^2)$：数据范围是整个实数轴，$\mu$、$\sigma^2$ 出现在密度公式里 $\rightarrow$ 用 $\bar{X}$、$s^2$
  \item $P(\lambda)$：$\lambda$ 出现在概率公式里，不影响范围 $\rightarrow$ 用 $\bar{X}$
\end{itemize}
\begin{dxtips}
    设 $X_1, X_2, \cdots, X_8$ i.i.d. $\sim U(0, \theta)$，
观测值为 $1, 2, 3, 4, 5, 6, 7, 8$，
则 $X_{(8)} = \max\{X_1, \cdots, X_8\} = 8$，
用 $X_{(8)}$ 估计 $\theta$。
我们求 $X_{(n)}$ 的期望来估计 $\theta$，
就像我们拿 $\bar{X}$ 的期望估计正态分布的 $\mu$。
求期望里面都是变量不是真实的数字是实验前。
\end{dxtips}
\begin{example}
$X_{(n)}$ 虽然是从 $X_1,\dots,X_8$ 里来的，但它不是随机抽的一个，
它是特意挑出最大的那个。挑最大值这个动作改变了分布。

具体来说，设 $\theta = 10$：

$$P(X_i \leq 5) = \frac{5}{10} = 0.5$$

随便一个 $X_i$ 有 $50\%$ 概率 $\leq 5$，合理。

$$P(X_{(8)} \leq 5) = \left(\frac{5}{10}\right)^8 = \frac{1}{256} \approx 0.4\%$$

8 个数里最大值还 $\leq 5$，概率只有 $0.4\%$。

直觉：8 个数都取到 5 以下，这件事几乎不可能发生，
因为总会有某个数跑到 5 以上。
所以 $X_{(n)}$ 虽然也是 $X_i$ 里的一个，
但它天生偏大，分布和普通 $X_i$ 不一样。
\end{example}
\begin{dxtips}
    这里的 $X_1, \dots, X_8$ 不是最大值 $X_{(8)}$，而是八个原始样本。
最大值小于 $5$ 等于全部八个样本都小于 $5$ 的概率，
每一个都服从 $U(0,\theta)$，小于 $5$ 的概率都是 $\dfrac{5}{\theta}$。
\end{dxtips}
\begin{note}
$$F_{(n)}(x) = \left(\frac{x}{\theta}\right)^n$$

$$f_{(n)}(x) = \frac{nx^{n-1}}{\theta^n}$$

$$E[X_{(n)}] = \int_0^\theta x \cdot \frac{nx^{n-1}}{\theta^n}\,dx 
= \frac{n}{\theta^n}\int_0^\theta x^n\,dx 
= \frac{n}{\theta^n} \cdot \frac{\theta^{n+1}}{n+1} 
= \frac{n\theta}{n+1}$$
\end{note}

\begin{dxtips}
    原本从 $0$ 积分到 $\theta$，故
$$E[X_{(n)}] = \frac{n\theta + 0}{n+1}$$
现从 $\theta_1$ 积分到 $\theta_2$，把 $\theta$ 换成 $\theta_2$，$0$ 换成 $\theta_1$：
$$E[X_{(n)}] = \frac{n\theta_2 + \theta_1}{n+1}$$

同理，$E[X_{(1)}]$ 把积分区域反过来，从 $\theta_2$ 积分到 $\theta_1$，
$n$ 后面跟着的是 $\theta_1$，另外加的是起始点 $\theta_2$：
$$E[X_{(1)}] = \frac{n\theta_1 + \theta_2}{n+1}$$
\end{dxtips}
\begin{note}

参数在密度公式里 $\rightarrow$ 用 $\bar{X}$、$s^2$ 这类自然统计量；参数在数据的取值范围上 $\rightarrow$ 用极值 $X_{(1)}$、$X_{(n)}$。
\end{note}
\begin{note}[顺序统计量期望的等间距直觉]
$n$ 个点均匀撒在 $[\theta_1, \theta_2]$ 上，平均来说这 $n$ 个点会把区间切成 $n+1$ 段等长的小段，每段长度为：
$$\frac{\theta_2 - \theta_1}{n+1}$$

因此：
\begin{itemize}
  \item $X_{(1)}$ 平均落在距离 $\theta_1$ 一段的位置：
  $$E[X_{(1)}] = \theta_1 + \frac{\theta_2-\theta_1}{n+1} = \frac{n\theta_1+\theta_2}{n+1}$$

  \item $X_{(n)}$ 平均落在距离 $\theta_2$ 一段的位置：
  $$E[X_{(n)}] = \theta_2 - \frac{\theta_2-\theta_1}{n+1} = \frac{\theta_1+n\theta_2}{n+1}$$
\end{itemize}

$X_{(1)}$ 是最小值，靠近 $\theta_1$，故 $\theta_1$ 权重大（$n$ 份），$\theta_2$ 权重小（$1$ 份）；$X_{(n)}$ 反之。

记忆：$n$ 个点把区间切成 $n+1$ 段，最小值在第 $1$ 段末，最大值在第 $n$ 段末。
\end{note}
\end{note}
\section{抽样分布，卡方，t，F}

\begin{definition}[$\chi^2$ 分布]
设 $X_1, \ldots, X_n \overset{\text{i.i.d.}}{\sim} N(0,1)$，令
$$\xi = \sum_{i=1}^n X_i^2$$
则称 $\xi$ 的分布为自由度为 $n$ 的 $\chi^2$ 分布，记作 $\xi \sim \chi^2(n)$。
\end{definition}

\begin{note}[要背的公式]
$$E(\xi) = n, \qquad \mathrm{Var}(\xi) = 2n.$$
\end{note}
\begin{dxtips}
    $\xi$念做可塞，是平方和。他的期望是因为E（X）都是0只有方差起效果每一个点方差都是1加起来就是n。就是不要想的太复杂因为就是01正态分布算一个是很好算把具体数字带进去就行了
\end{dxtips}
\begin{note}
方差定义变形：$\mathrm{Var}(X) = E(X^2) - [E(X)]^2$，移项得 $E(X^2) = \mathrm{Var}(X) + [E(X)]^2$。
\end{note}
\begin{note}[Var$(\xi)=2n$ 的推导]
由独立可加性：
$$\mathrm{Var}(\xi) = \sum_{i=1}^n \mathrm{Var}(X_i^2)$$
对单个 $X_i \sim N(0,1)$：
$$\mathrm{Var}(X_i^2) = E(X_i^4) - [E(X_i^2)]^2 = 3 - 1 = 2$$
其中 $E(X^4)=3$ 是标准正态的四阶矩（直接用）。故 $\mathrm{Var}(\xi) = 2n$。
\end{note}

\begin{theorem}[可加性]
设 $\xi \sim \chi^2(m)$，$\eta \sim \chi^2(n)$，且 $\xi \perp \eta$，则
$$\xi + \eta \sim \chi^2(m+n).$$
\end{theorem}

\begin{note}[自由度的直觉]
有几个独立的标准正态变量参与平方求和，自由度就是几。一旦存在约束（例如 $\sum_{i=1}^n(X_i - \bar{X}) = 0$），自由度减一——这就是样本方差给出 $\chi^2(n-1)$ 而非 $\chi^2(n)$ 的原因。
\end{note}
\begin{definition}[$t$ 分布--标准正态/根号卡方分布/自由度]
设 $X \sim N(0,1)$，$Y \sim \chi^2(n)$，且 $X \perp Y$，令
$$\xi = \frac{X}{\sqrt{Y/n}}$$
则称 $\xi$ 的分布为自由度为 $n$ 的 $t$ 分布，记作 $\xi \sim t(n)$。
\end{definition}
\begin{dxtips}
    所以看见t分布得想谁提供标准正态分布谁提供卡方分布
\end{dxtips}

\begin{note}[为什么要构造 $t$ 分布]
实际中想算标准化统计量 $\dfrac{\bar{X}-\mu}{\sigma/\sqrt{n}}$，但 $\sigma$ 未知，用样本标准差 $S$ 替换后分母变成随机变量，不再服从正态分布。这个比值的精确分布就是 $t$ 分布。
\end{note}
\begin{dxtips}
$\sigma$ 已知时分母是常数，整体服从正态分布，用 $Z$ 检验。$\sigma$ 未知时引入样本标准差 $S$，随着 $S$ 被引进，卡方分布也被引进（因为 $\frac{(n-1)S^2}{\sigma^2}\sim\chi^2(n-1)$），正态除以根号卡方，整体变成 $t$ 分布。
\end{dxtips}
\begin{note}[渐近性质]
$$t(n) \xrightarrow{n \to \infty} N(0,1)$$
样本量充分大时，$t$ 分布趋向标准正态。
\end{note}
\begin{definition}[$F$ 分布]
设 $X \sim \chi^2(m)$，$Y \sim \chi^2(n)$，且 $X \perp Y$，令
$$\xi = \frac{X/m}{Y/n}$$
则称 $\xi$ 的分布为自由度为 $(m,n)$ 的 $F$ 分布，记作 $\xi \sim F(m,n)$。
\end{definition}
\begin{dxtips}
    F分布本质上描述的事两个卡方分布的关系，只不过他们都要处以自己的自由度罢了。自由度分子在前分母在后。需要找到两个卡方才能用F分布
\end{dxtips}
\begin{note}[倒数性质]
$$\xi \sim F(m,n) \implies \frac{1}{\xi} \sim F(n,m)$$
\end{note}

\begin{note}[分位数关系]
$$F_{1-\alpha}(m,n) = \frac{1}{F_{\alpha}(n,m)}$$
表中只给出 $F_\alpha$，遇到 $F_{1-\alpha}$ 用此关系换算。
\end{note}

\subsection{卡方t与伽马函数}
\begin{note}
密度函数的统一结构：\(f(x)=\text{归一化系数}\times\text{核（决定形状的部分）}\)。

\textbf{$\chi^2(n)$：}
\[
f(x)=\underbrace{\frac{1}{2^{n/2}\Gamma(n/2)}}_{\text{归一化系数}}
\times
\underbrace{x^{\frac n2-1}e^{-x/2}}_{\text{核}},
\qquad x>0.
\]

\textbf{$t(n)$：}
\[
f(x)=\underbrace{\frac{\Gamma\!\left(\frac{n+1}{2}\right)}{\sqrt{n\pi}\,\Gamma\!\left(\frac n2\right)}}_{\text{归一化系数}}
\times
\underbrace{\left(1+\frac{x^2}{n}\right)^{-\frac{n+1}{2}}}_{\text{核}}.
\]
\end{note}
\begin{note}
\[
N(0,1):\quad \varphi(x)=\underbrace{\frac{1}{\sqrt{2\pi}}}_{\text{归一化}}\cdot\underbrace{e^{-x^2/2}}_{\text{核}},
\qquad x\in\mathbb{R}.
\]

\[
\chi^2(1):\quad f(x)=\underbrace{\frac{1}{\sqrt{2\pi}}}_{\text{归一化（一模一样！）}}\cdot\underbrace{x^{-1/2}e^{-x/2}}_{\text{核}},
\qquad x>0.
\]

\[
\chi^2(n):\quad f(x)=\underbrace{\frac{1}{2^{n/2}\Gamma(\frac n2)}}_{\text{归一化}}\cdot\underbrace{x^{\frac n2-1}e^{-x/2}}_{\text{核}},
\qquad x>0.
\]
\end{note}
\begin{dxtips}
整个 \(\Gamma\) 函数最重要的就是 \(x\) 的头上是什么，而不是 \(e\) 的头上是什么——\(x\) 头上是几，加1就是分母 \(\Gamma\) 了嘛的东西，和 \(2\) 的次方。伽马展开其实就是一个积分的式子对吗，只不过积分式子长得太长了不好写，正好我们带入每一个伽马都能用另一个公式算出整个积分的值。其实和我们要弄成标准正态好查表一样弄成标准伽马我们也是好计算
\end{dxtips}
\begin{dxtips}
换元 \(x=2u\) 的目标只有一个——让 \(e\) 头上变成单纯的 \(-u\)，这样才能跟 \(\Gamma(z)=\displaystyle\int_0^\infty u^{z-1}e^{-u}\,du\) 这个"标准模板"对上号。因为x变成2u所以u头上自然继承了x头上的指数，由于dx变成2du所以他会比x头上指数多一次
\end{dxtips}
\begin{dxtips}
\(\Gamma(1/2)\) 与高斯积分本质上是同一个积分，换元 \(x=u^2\) 即可互相转化：

令 \(x=u^2\)（\(u>0\)），则 \(dx=2u\,du\)，\(x^{-1/2}=u^{-1}\)，代入：
\[
\Gamma\!\left(\tfrac12\right)=\int_0^\infty x^{-1/2}e^{-x}\,dx
=\int_0^\infty u^{-1}e^{-u^2}\cdot(2u\,du)
=\int_0^\infty 2e^{-u^2}\,du.
\]
再由 \(e^{-u^2}\) 为偶函数，\(2\int_0^\infty=\int_{-\infty}^{\infty}\)，得
\[
\Gamma\!\left(\tfrac12\right)=\int_{-\infty}^{\infty}e^{-u^2}\,du=\sqrt\pi.
\]

这个换元 \(x=u^2\) 正是 \(Y=X^2\)（把 \(N(0,1)\) 变成 \(\chi^2(1)\)）在积分层面的体现——所以 \(\chi^2(1)\) 和 \(N(0,1)\) 的归一化系数都是 \(1/\sqrt{2\pi}\)，本质上是同一个积分换了个变量名。
\end{dxtips}
\begin{note}
\textbf{$\Gamma$函数三条拆解规则：}
\begin{itemize}
    \item \(\Gamma(z+1)=z\,\Gamma(z)\)：递推（分部积分得到，本质是阶乘"往下乘"）
    \item \(\Gamma(n)=(n-1)!\)：\(n\) 为正整数时，纯阶乘，不含 \(\pi\)
    \item \(\Gamma\!\left(\tfrac12\right)=\sqrt\pi\)：唯一的"魔法常数"，来自高斯积分 \(\displaystyle\int_{-\infty}^{\infty}e^{-x^2}dx=\sqrt\pi\)
\end{itemize}

\end{note}
\begin{dxtips}
    伽马函数本质是一个积分，对积分里面的x求导指数下来变成前面的系数也就是z，指数减去1但是z-1的为指数伽马就是z
\end{dxtips}
\begin{note}
\(\Gamma(z+1)=z\Gamma(z)\) 这条递推，来自分部积分——本质是"幂函数求导会掉一个指数下来，指数函数积分还是自己"：
\[
\Gamma(z+1)=\int_0^\infty u^z e^{-u}\,du
\overset{\text{分部积分}}{=}
\Bigl[-u^z e^{-u}\Bigr]_0^\infty+z\int_0^\infty u^{z-1}e^{-u}\,du
=z\,\Gamma(z).
\]
对 \(u^z\) 求导得 \(z\,u^{z-1}\)，对 \(e^{-u}\) 积分还是 \(-e^{-u}\)（自我复制），边界项为0，分部积分直接给出 \(\Gamma(z+1)=z\,\Gamma(z)\)。

这正是 Euler 当年的 motivation：\(n!=n\cdot(n-1)\cdots1\) 这种"阶乘递推"，能不能找一个对所有实数 \(z\) 都成立的"连续版"？答案就是这个积分——\(\Gamma(z)\) 是阶乘往非整数方向的"插值"。
\end{note}
\begin{note}
（可选，更大的图景）"升维+极坐标"能推广到n维——\(\chi^2(n)\) 本质上是"n维标准正态向量的模长平方"，\(\Gamma(n/2)\) 正是n维球体积/表面积公式里的那个 \(\Gamma\)。所以 \(\Gamma(n/2)\) 出现在 \(\chi^2(n)\) 里不是巧合，而是n维球几何的直接体现——这部分对考试不直接有用，但解释了"为什么偏偏是\(\Gamma(n/2)\)"这个终极问题。
\end{note}
\section{正态三定理}
\begin{note}[正态总体三大定理]
设 $X_1, \ldots, X_n \overset{\text{i.i.d.}}{\sim} N(\mu, \sigma^2)$。

\textbf{定理一：}
$$\bar{X} \sim N\!\left(\mu,\, \frac{\sigma^2}{n}\right), \qquad \frac{(n-1)s^2}{\sigma^2} \sim \chi^2(n-1), \qquad \bar{X} \perp s^2$$

\textbf{定理二（$\sigma^2$ 未知时估计 $\mu$）：}
$$T = \frac{\bar{X}-\mu}{s/\sqrt{n}} \sim t(n-1)$$

\textbf{定理三（两独立正态总体，$Y_1,\ldots,Y_m \overset{\text{i.i.d.}}{\sim} N(\mu_2,\sigma_2^2)$）：}
$$F = \frac{s_1^2/\sigma_1^2}{s_2^2/\sigma_2^2} \sim F(n-1,\, m-1)$$
\end{note}
\begin{definition}[样本方差]
$$s^2 = \frac{1}{n-1}\sum_{i=1}^n (X_i - \bar{X})^2$$
分母用 $n-1$ 而非 $n$，使得 $E(s^2) = \sigma^2$（无偏）。
\end{definition}
\begin{dxtips}
    乘n-1就变成了μ等于0的正态分布除以σ平方就是变成了标准正态分布，
\end{dxtips}
\begin{note}[定理二的推导]
$$T = \frac{\bar{X}-\mu}{s/\sqrt{n}} = \frac{\dfrac{\bar{X}-\mu}{\sigma/\sqrt{n}}}{\sqrt{\dfrac{(n-1)s^2/\sigma^2}{n-1}}} = \frac{N(0,1)}{\sqrt{\chi^2(n-1)/(n-1)}} \sim t(n-1)$$
分子是标准正态，分母是 $\chi^2(n-1)$ 除以自由度再开根，$\sigma^2$ 上下消掉，对上 $t$ 分布定义。
\end{note}
\begin{dxtips}
    根号n是给分子做标准化的分母本来就是定理一段样子因为上下都有σ平方消l由于算t分布卡方分布得除以自己的自由度再开方所以n-1项也没有了
\end{dxtips}
\begin{note}[为什么自由度是 $n-1$]
求和有 $n$ 项，但存在约束
$$\sum_{i=1}^n (X_i - \bar{X}) = 0$$
知道前 $n-1$ 项，最后一项被锁死。真正能自由变化的只有 $n-1$ 个，故自由度为 $n-1$。

\textbf{$n$ 个数，$1$ 个约束，剩 $n-1$ 个自由度。}
\end{note}
\begin{note}[总体方差与样本方差对比]

\begin{itemize}
  \item \textbf{总体方差：} $\sigma^2 = E[(X-\mu)^2]$，用真实均值 $\mu$，是分布的参数，未知。
  \item \textbf{样本方差：} $s^2 = \dfrac{1}{n-1}\displaystyle\sum_{i=1}^n(X_i-\bar{X})^2$，用样本均值 $\bar{X}$，是统计量，可计算。
  一个那E来求加权平均一个直接用/n-1求平均
\end{itemize}
\end{note}
\subsection{前置引理}
\begin{lemma}[正交变换引理]
设 $X_1, \ldots, X_n$ 相互独立，$X_i \sim N(\mu_i, \sigma^2)$（方差相同），$A$ 为 $n \times n$ 正交矩阵（$A^\top A = I$），令 $Y = AX$，则 $Y_1, \ldots, Y_n$ 相互独立，且
$$Y_i \sim N\!\left(\sum_k a_{ik}\mu_k,\ \sigma^2\right)$$
\end{lemma}
\begin{dxtips}
每个 $X_i$ 都是一个数，就是第 $i$ 次观测的结果。向量 $\boldsymbol{X} = (X_1, X_2, \ldots, X_n)^\top$ 只是把 $n$ 次观测的数堆成一列，方便矩阵运算。
\end{dxtips}
\begin{dxtips}
    左乘一个矩阵的数学意义是换成左乘的坐标系,而最后的结果会变成右边的维度，所以左乘可以维度大于本身。
    [列看法（Strang）]
$AX$ 是 $A$ 各列的线性组合，$X$ 的分量是权重：
$$AX = x_1\mathrm{col}_1(A) + x_2\mathrm{col}_2(A) + \cdots + x_n\mathrm{col}_n(A)$$
$AX$ 的结果落在 $A$ 的列空间里。

\end{dxtips}
\begin{example}[列看法具体计算]
$$A = \begin{pmatrix}1&2\\2&1\end{pmatrix}, \quad X = \begin{pmatrix}3\\4\end{pmatrix}$$
因为运算的时候是行乘列，作揖3这个位置只有每行的第一个位置能和他乘每一行的第一个位置组合起来就是第一列，所以3就是第一列的权重同样4就是第二列。
$$AX = 3\begin{pmatrix}1\\2\end{pmatrix} + 4\begin{pmatrix}2\\1\end{pmatrix} = \begin{pmatrix}3\\6\end{pmatrix} + \begin{pmatrix}8\\4\end{pmatrix} = \begin{pmatrix}11\\10\end{pmatrix}$$
\end{example}
\begin{note}[正交变换的几何意义]
左乘正交矩阵就是在 $n$ 维空间里做旋转（或翻转），把原来的坐标系转到新的坐标系。正交矩阵的旋转保长度——向量的模不变，所以散布程度（方差）也不变，这就是为什么 $Y$ 的每个分量方差还是 $\sigma^2$。
\end{note}
\begin{theorem}[柯赫伦定理]
设 $X_1, \ldots, X_n \overset{\text{i.i.d.}}{\sim} N(0,1)$，总平方和分解为
$$\sum_{i=1}^n X_i^2 = Q_1 + Q_2 + \cdots + Q_k$$
其中 $Q_i = \boldsymbol{X}^\top A_i \boldsymbol{X}$ 为二次型，$\mathrm{rank}(A_i) = n_i$。则 $Q_1, \ldots, Q_k$ 相互独立且各自 $\sim \chi^2(n_i)$ 的充要条件为
$$n_1 + n_2 + \cdots + n_k = n$$
\end{theorem}

\begin{note}[直觉]
总自由度 $n$ 被完整拆分给每一块，没有浪费也没有重叠，每块才能独立地服从卡方分布。
\end{note}
\section{柯西分布}\begin{definition}[标准柯西分布]
设 $X,Y\overset{\text{i.i.d.}}{\sim} N(0,1)$，令
$$\xi=\frac{X}{Y}$$
则称 $\xi$ 的分布为标准柯西分布，记作 $\xi\sim C(0,1)$，密度函数为
$$f(x)=\frac{1}{\pi(1+x^2)}.$$
\end{definition}

\begin{note}[柯西与 $t$ 分布的关系：$t(1)=C(0,1)$]
回顾 $t$ 分布定义 $\xi=\dfrac{X}{\sqrt{Y/n}}$，$X\sim N(0,1)$，$Y\sim\chi^2(n)$。

取 $n=1$：$Y\sim\chi^2(1)$ 即 $Y=Z^2$（$Z\sim N(0,1)$），于是
$$\xi=\frac{X}{\sqrt{Z^2}}=\frac{X}{|Z|}.$$
由 $X$ 关于 $0$ 对称且与 $Z$ 独立，$X/|Z|$ 与 $X/Z$ 同分布，故
$$t(1)=C(0,1).$$
\end{note}

\begin{dxtips}
柯西不是一个孤立的分布，它是 $t$ 分布谱系的左端点（$n=1$），正态是右端点（$n\to\infty$，已在"渐近性质"里）。中间由自由度 $n$ 连续过渡，"谁提供标准正态、谁提供卡方"的框架完全通用，只是 $n=1$ 时退化成柯西。
\end{dxtips}

\begin{note}[最大的区别：柯西没有矩]
\begin{itemize}
  \item 正态：$f(x)=\dfrac{1}{\sqrt{2\pi}}e^{-x^2/2}$，尾部指数衰减，所有阶矩都存在。
  \item 柯西：$f(x)=\dfrac{1}{\pi(1+x^2)}$，尾部只有 $1/x^2$ 衰减（"厚尾"），
  $$\int_{-\infty}^{\infty}|x|\cdot\frac{1}{\pi(1+x^2)}\,dx$$
  在 $\pm\infty$ 处的发散速度与 $\displaystyle\int\frac{1}{x}\,dx$ 相同 $\implies E(X)$ 不存在，方差自然也不存在。
\end{itemize}
\end{note}

\begin{dxtips}
特征函数上看更直接（联系 44(6)）：正态特征函数 $e^{-t^2/2}$ 在 $t=0$ 处光滑可导；柯西特征函数 $e^{-|t|}$ 在 $t=0$ 处有"尖点"不可导。特征函数在 $0$ 点的导数对应矩，不可导 $\iff$ 矩不存在——这就是柯西"差"在哪。
\end{dxtips}

\begin{note}[大数定律失效：$\bar X$ 的行为对比]
\begin{center}
\begin{tabular}{c|c|c}
 & $\bar X$ 的分布 & 随 $n$ 变化 \\\hline
正态 $N(\mu,\sigma^2)$ & $N(\mu,\sigma^2/n)$ & 方差 $\to 0$，$\bar X\to\mu$（LLN 成立） \\
柯西 $C(\mu,\gamma)$ & $C(\mu,\gamma)$ & 形状不变，不收缩，LLN 失效
\end{tabular}
\end{center}
LLN 需要"均值存在"作为前提，柯西连均值都没有，自然谈不上"收敛到均值"。
\end{note}

\begin{note}[尺度性质]
正态的缩放作用在\textbf{方差}上，按 $a^2$ 走：
$$aX\sim N(a\mu,\,a^2\sigma^2).$$
柯西的缩放作用在\textbf{尺度参数}上，按 $|a|$ 走（一次方）：
$$aC(0,1)=C(0,a).$$
柯西没有"方差"概念，$\gamma$ 只是描述分布"胖瘦"的尺度参数。
\end{note}

\begin{dxtips}
做题识别口诀：看见"两个独立标准正态相除"或者"$t(1)$" $\to$ 想到柯西 $\to$ 矩不存在、LLN 失效、缩放是一次方不是平方。
\end{dxtips}
 \begin{definition}[一般柯西分布 $C(\mu,\gamma)$]
$$f(x)=\frac{\gamma}{\pi\big(\gamma^2+(x-\mu)^2\big)}$$
其中 $\mu$ 为位置参数，$\gamma>0$ 为尺度参数。取 $\mu=0,\gamma=1$ 即标准柯西 $f(x)=\dfrac{1}{\pi(1+x^2)}$。
\end{definition}

\begin{dxtips}
$C(\mu,\gamma)$ 里：
\begin{itemize}
  \item $\mu$ 是位置参数（location），决定分布"中心"在哪——密度峰值（最高点）就在 $x=\mu$ 处，类比正态的均值 $\mu$。
  \item $\gamma$ 是尺度参数（scale），决定分布"胖瘦/宽窄"，类比正态的 $\sigma$（不是 $\sigma^2$）。
\end{itemize}

直观理解 $\gamma$：密度在 $x=\mu$ 处最高，下降到"峰值的一半"时，对应的 $x$ 离 $\mu$ 的距离就是 $\gamma$——所以 $\gamma$ 叫"半高宽"（half-width at half-maximum）。$\gamma$ 越大，分布越扁越宽；$\gamma$ 越小，分布越尖越窄。
\end{dxtips}
\chapter{能力一}
\section{矩估计}
\begin{note}
参数估计的根本困境：$\theta$ 活在"理论世界"里，数据 $X_1,\dots,X_n$ 活在"经验世界"里，两者之间没有直接的桥。

矩估计法找到的桥就是"矩"——它同时存在于两个世界：

\begin{itemize}
    \item \textbf{理论世界}：$E(X^k)=g_k(\theta)$，只要分布族确定了，它就是 $\theta$ 的一个函数（已知形式，含未知数 $\theta$）。
    \item \textbf{经验世界}：样本矩 $\dfrac1n\displaystyle\sum_{i=1}^n X_i^k$，纯粹由数据算出来的数，不需要知道 $\theta$ 是什么。
\end{itemize}

大数定律（LLN）保证：当 $n\to\infty$ 时，
\[
\frac1n\sum_{i=1}^n X_i^k \xrightarrow{P} E(X^k).
\]
也就是"经验世界"算出来的这个数，会收敛到"理论世界"那个含 $\theta$ 的数——一个能算、一个含未知数，但本质上是同一个量的两种表达。

矩估计法就是\textbf{强行让两者相等}：
\[
g_k(\theta)=\frac1n\sum_{i=1}^n X_i^k,
\]
再把这个方程反解出 $\theta$。这一步"反解"，就是从经验世界跨回理论世界、解出 $\theta$ 的那座桥。
\end{note}
\begin{dxtips}
一堆数字，我知道它们服从一个特定的分布，比如泊松分布 $P(\lambda)$，但直接看不出来 $\lambda$ 是多少。所以我们需要找一个含有 $\lambda$ 的公式，同时这个公式也能用真实数据算出来——含有 $\lambda$ 的公式，我们选取的是 $\lambda$ 的（总体）矩 $E(X)=\lambda$；能用真实数据算的公式，则是样本矩 $\bar X=\dfrac1n\sum_{i=1}^n X_i$。
    \end{dxtips}
    \begin{note}
$\hat{\theta}$ 中的 hat 表示\textbf{对 $\theta$ 的点估计}，不特指某种方法，任何估计方法给出的点估计都可以戴 hat。

\begin{center}
\begin{tabular}{lll}
\toprule
\textbf{方法} & \textbf{写法} & \textbf{本质} \\
\midrule
MLE & $\hat{\theta}_{\mathrm{MLE}}$ & 最大化似然 \\
矩估计 & $\hat{\theta}_{M}$ & 样本矩 $=$ 总体矩 \\
贝叶斯估计 & $\hat{\theta}_{B}$ & 后验期望 $E(\theta\mid\boldsymbol{X})$ \\
\bottomrule
\end{tabular}
\end{center}

hat 表示身份（点估计），不表示计算方式。
\end{note}
\section{MLE}
\begin{definition}[最大似然估计]
设 $X_1,\ldots,X_n$ 是来自总体 $f(x;\theta)$（$\theta\in\Theta$）的简单随机样本，$x_1,\ldots,x_n$ 为观测值。称
\[
L(\theta)=\prod_{i=1}^n f(x_i;\theta)
\]
为似然函数。若存在 $\hat\theta\in\Theta$ 使得
\[
\hat\theta=\arg\max_{\theta\in\Theta}L(\theta),
\]
则称 $\hat\theta$ 为 $\theta$ 的最大似然估计（MLE）。
\end{definition}
\begin{dxtips}
$f(x;\theta)$ 这同一个公式：固定 $\theta$ 看 $x$ 怎么变 $\to$ 叫"概率"；固定 $x$（已观测数据）看 $\theta$ 怎么变 $\to$ 叫"似然"——同一个表达式，换了阅读方向。

$\hat\theta$ 不是似然函数 $L(\theta)$ 的最大值本身，而是\textbf{取得最大值时对应的 $\theta$}

"最优" $=$ 让\textbf{当前这组数据}最可能发生的 $\theta$，是基于这组样本的估计值，不是真实 $\theta$；换一组样本，$\hat\theta$ 会变，所以它本身是随机变量。
\end{dxtips}
\subsection{如何算mle}
\begin{definition}[支撑集]
对于随机变量 $X$ 的密度函数（或概率质量函数）$f(x;\theta)$，称集合
\[
\mathrm{supp}(f)=\{x\in\mathbb{R}: f(x;\theta)>0\}
\]
为其支撑集——即该分布真正可能取到的 $x$ 值的全体；在支撑集之外，$f(x;\theta)=0$。

\begin{itemize}
    \item $N(\mu,\sigma^2)$：支撑集 $=\mathbb{R}$ —— 不依赖参数
    \item $U(0,\theta)$：支撑集 $=[0,\theta]$ —— \textbf{依赖 $\theta$}
\end{itemize}
\end{definition}
\begin{note}[MLE 的两种算法：看 $\theta$ 是否出现在支撑集里]
\begin{itemize}
    \item \textbf{$\theta$ 不出现在支撑集里}（支撑集不依赖 $\theta$，如 $N(\mu,\sigma^2)$、$\mathrm{Exp}(\lambda)$、泊松等）：

    $f(x;\theta)$ 关于 $\theta$ 处处光滑，直接用\textbf{求导法}：
    \[
    \ell(\theta)=\sum_{i=1}^n\ln f(x_i;\theta),\qquad \ell'(\theta)=0\ \Longrightarrow\ \hat\theta.
    \]
    解似然方程即可（必要时验证 $\ell''(\hat\theta)<0$）。

    \item \textbf{$\theta$ 出现在支撑集里}（支撑集依赖 $\theta$，如 $U(0,\theta)$、两参数指数 $x\ge\mu$）：

    $L(\theta)$ 含指示函数 $\mathbb{1}[\cdots]$，求导法通常无解。改用\textbf{边界分析}：
    \begin{enumerate}
        \item \textbf{可行域}：把支撑集条件翻译成"$\theta$ 必须满足的范围"（由观测数据反过来限制 $\theta$）
        \item \textbf{单调性}：在可行域内看 $L(\theta)$ 关于 $\theta$ 是增还是减
        \item \textbf{边界}：$\hat\theta$ 取可行域中朝增长方向的那一端（通常是某个顺序统计量 $X_{(1)}$ 或 $X_{(n)}$）
    \end{enumerate}
\end{itemize}
\end{note}
\begin{note}[MLE 的不变性]
若 $\hat\theta$ 是 $\theta$ 的MLE，则对任意函数 $g$，$g(\hat\theta)$ 就是 $g(\theta)$ 的MLE：
\[
\widehat{g(\theta)}_{\mathrm{MLE}}=g\big(\hat\theta_{\mathrm{MLE}}\big).
\]

直觉：似然函数 $L(\theta)$ 的峰在 $\theta=\hat\theta$ 处；把横轴从 $\theta$ 换成 $\eta=g(\theta)$，只是给同一个峰换了坐标标签，峰的位置不变——只是从 $\hat\theta$ 变成 $g(\hat\theta)$。

例：$X\sim\mathrm{Exp}(\lambda)$ 的 MLE 为 $\hat\lambda=1/\bar X$，则总体均值 $1/\lambda$ 的MLE直接是 $1/\hat\lambda=\bar X$，不需要重新求解。
\end{note}
\section{贝叶斯}
\subsection{后验}
\begin{note}
贝叶斯的世界里，$\theta$ 不是固定的未知数，而是一个\textbf{随机变量}，有自己的分布。

\begin{table}[h]
\centering
\begin{tabular}{lll}
\toprule
\textbf{名称} & \textbf{符号} & \textbf{含义} \\
\midrule
先验分布 & $\pi(\theta)$ & 观测数据\textbf{之前}，你对 $\theta$ 的信念 \\
似然函数 & $f(X\mid\theta)$ & 给定 $\theta$，数据出现的概率 \\
后验分布 & $p(\theta\mid X)$ & 看到数据\textbf{之后}，更新后的信念 \\
\bottomrule
\end{tabular}
\end{table}
\end{note}
\begin{note}[MLE 与 Bayes 的关系]
两者目标相同——都给出 $\theta$ 的估计值，区别在于"用什么信息、怎么组合"：
\begin{itemize}
  \item MLE：只用数据，找"最能解释数据"的 $\theta$
  \item Bayes：先验(预想) + 数据 $\to$ \textbf{融合}成新估计
\end{itemize}

Bayes 不是"判断预想对不对"（假设检验式的二元框架），而是把先验和数据按可信度\textbf{加权融合}出一个新数值：硬币例子中先验 $p\approx0.5$，数据显示 $\approx0.8$，融合后 $\hat p\approx0.7$——数据越多越靠近数据，先验越强越靠近先验。
\end{note}
\begin{example}
抛硬币 $n=10$ 次，观测正面 $k=7$ 次，估计正面概率 $\theta$。
设先验 $\pi(\theta)\sim\mathrm{Beta}(2,2)$，即 $\pi(\theta)\propto\theta(1-\theta)$。

似然函数为
\[
f(X\mid\theta)=\binom{10}{7}\theta^7(1-\theta)^3\propto\theta^7(1-\theta)^3.
\]

后验分布为
\[
p(\theta\mid X)\propto\theta^7(1-\theta)^3\cdot\theta(1-\theta)=\theta^8(1-\theta)^4,
\]
认出这是 $\mathrm{Beta}(9,5)$，故 Bayes 估计（后验均值）为
\[
\hat{\theta}=E[\theta\mid X]=\frac{9}{9+5}=\frac{9}{14}\approx 0.643.
\]
相比频率估计 $7/10=0.7$，先验将结果向 $0.5$ 拉近了一点。
\end{example}
\begin{note}
在贝叶斯推导中，后验分布的完整表达式为
\[
p(\mu|\boldsymbol{X}) = \frac{f(\boldsymbol{X}|\mu)\cdot\pi(\mu)}{f(\boldsymbol{X})}
\]
其中分母 $f(\boldsymbol{X}) = \int f(\boldsymbol{X}|\mu)\,\pi(\mu)\,d\mu$ 对 $\mu$ 而言是常数，
故只需保留含 $\mu$ 的部分：
\[
p(\mu|\boldsymbol{X}) \propto f(\boldsymbol{X}|\mu)\cdot\pi(\mu)
\]
这一步的合理性在于：概率密度满足归一化条件 $\int p(\mu|\boldsymbol{X})\,d\mu = 1$，
因此\textbf{一旦认出核函数的形状，归一化常数即被唯一确定}，无需显式计算。

例如，配方后若发现
\[
p(\mu|\boldsymbol{X}) \propto \exp\!\left(-\frac{(\mu-\mu_1)^2}{2\tau_1^2}\right)
\]
则立即断定后验为 $N(\mu_1,\tau_1^2)$，归一化常数 $\dfrac{1}{\sqrt{2\pi}\,\tau_1}$ 自动补齐。

\textbf{本质：}$\propto$ 不是"不关心常数"，而是"先认出分布族，再补常数"。
此技巧在共轭先验推导中极为常用。
\end{note}
\subsection{beta函数}
\begin{definition}{Beta 函数}{}
Beta 函数（大写 $B$）定义为
\[
B(\alpha,\beta) = \int_0^1 t^{\alpha-1}(1-t)^{\beta-1}\,dt, \quad \alpha,\beta>0.
\]
它本身是一个数，用来对 Beta 分布做归一化。例如
$B(2,2)=\int_0^1 t(1-t)\,dt=\dfrac{1}{6}$。

与 Gamma 函数的关系：
\[
B(\alpha,\beta)=\frac{\Gamma(\alpha)\Gamma(\beta)}{\Gamma(\alpha+\beta)}.
\]
\end{definition}
\begin{note}
\textbf{Beta 函数更深的意义}

\textbf{1.~它是 $[0,1]$ 上的 Gamma}

Gamma 函数管的是 $(0,\infty)$ 上的幂次积分：
\[
\Gamma(\alpha) = \int_0^\infty x^{\alpha-1}e^{-x}\,dx
\]
Beta 函数管的是 $(0,1)$ 上的幂次积分：
\[
B(\alpha,\beta) = \int_0^1 x^{\alpha-1}(1-x)^{\beta-1}\,dx
\]
$x$ 和 $1-x$ 加起来等于 $1$——这正是它天然描述``比例、概率''的原因。

\textbf{2.~它是组合数的连续推广}

注意二项式系数：
\[
\binom{n}{k} = \frac{n!}{k!(n-k)!} = \frac{1}{(n+1)\,B(k+1,\,n-k+1)}
\]
Beta 函数就是把离散的组合数\textbf{连续化}了。抛硬币用 Beta 先验，本质上是在做连续版本的计数。
\end{note}
\begin{dxtips}
$B(\alpha,\beta)$ 两个参数的含义：类比抛硬币，$\alpha$ 和 $\beta$ 可理解为\textbf{虚拟的历史数据}。

\begin{table}[h]
\centering
\begin{tabular}{lll}
\toprule
\textbf{参数} & \textbf{位置} & \textbf{含义} \\
\midrule
$\alpha$ & 第一个 & 历史上观测到正面的次数 $+1$ \\
$\beta$  & 第二个 & 历史上观测到反面的次数 $+1$ \\
\bottomrule
\end{tabular}
\end{table}

\begin{itemize}
    \item $\mathrm{Beta}(2,2)$：相当于"我之前\textbf{好像见过1次正面、1次反面}"，感觉差不多均匀。
    \item $\mathrm{Beta}(1,1)$：0次正面、0次反面——\textbf{什么都不知道}，即均匀分布 $U(0,1)$。
\end{itemize}

每次观测到新数据，$\alpha$ 或 $\beta$ 加一，后验就是把虚拟历史与真实数据\textbf{合并计数}，这就是共轭更新的本质。
\end{dxtips}
\begin{note}
\textbf{共轭更新}：先验与后验属于\textbf{同一族分布}，后验参数 $=$ 先验参数 $+$ 数据计数，无需计算积分。

以 Beta--Binomial 为例，观测到 $k$ 次正面、$n-k$ 次反面：
\[
\underbrace{\mathrm{Beta}(\alpha,\beta)}_{\text{先验}}
\xrightarrow{\text{观测}\ n\ \text{次}}
\underbrace{\mathrm{Beta}(\alpha+k,\ \beta+n-k)}_{\text{后验}}
\]
原因：似然与先验相乘后指数直接合并，
\[
\theta^{\alpha-1}(1-\theta)^{\beta-1}\cdot\theta^{k}(1-\theta)^{n-k}
=\theta^{(\alpha+k)-1}(1-\theta)^{(\beta+n-k)-1},
\]
形状仍是 Beta，故参数直接相加即得后验。
\end{note}

\textbf{第二步：推导后验参数，得贝叶斯估计量。}
\[
p(\theta|\boldsymbol{X}) \propto \theta^{S+a-1}(1-\theta)^{n-S+b-1}
\implies \theta|\boldsymbol{X} \sim \text{Beta}(S+a,\ n-S+b)
\]

\[
\hat{\theta}_B = \frac{S+a}{n+a+b}, \quad S = \sum_{i=1}^n X_i
\]

后续偏差、方差、MSE 均对 $S$ 求期望。
\end{solution}

\chapter{能力二}
\section{无偏估计}
很好的问题，这是入门时最容易懵的地方。

\textbf{估计量的定义就是：样本的任意函数。}

你有数据 $X_1, X_2, \ldots, X_n$，你想猜 $\mu$。只要是用这些数据算出来的一个数，就叫估计量，没有任何其他限制。

比如 $n=3$，数据是 $2, 4, 6$：

\begin{center}
\begin{tabular}{lll}
\hline
估计量 & 公式 & 算出来 \\
\hline
$\bar{X}$ & $(2+4+6)/3$ & $4$ \\
$X_1$ & 只取第一个 & $2$ \\
$(X_1+X_3)/2$ & $(2+6)/2$ & $4$ \\
常数 $5$ & 不管数据，直接报 $5$ & $5$ \\
$\max(X_i)$ & 取最大值 & $6$ \\
\hline
\end{tabular}
\end{center}

这五个都是"合法"的估计量——你都可以拿来交给别人说"我估计 $\mu = $ 这个数"。

\textbf{问题就在这里}：到底用哪个？

直觉上 $\bar{X}$ 最好，但"直觉"不是数学。能力二要做的事就是：给出有数学保证的标准，说清楚为什么 $\bar{X}$ 比 $X_1$ 好，好在哪里，好多少。

一句话总结：\textbf{估计量 $=$ 数据的函数，函数有无穷多个，所以估计量不唯一。}能力二就是在这无穷多个里面找"最好的"。
\begin{note}
    总结一句话：你只做一次实验，\textbf{估计量}是你选的公式，\textbf{估计值}是代入数据后算出的那个数。``多次实验''是数学上验证公式好不好用的思想实验，不是真的要做很多次。
\end{note}
\begin{note}
    \textbf{第一个：不知道 $\mu$ 怎么对准？}

``对准''不是你去瞄准一个已知的靶，而是通过代数推导证明一个必然成立的性质。

$X_1, X_2, \ldots, X_n$ 来自 $N(\mu, \sigma^2)$，所以 $E[X_i] = \mu$，于是：

$$E[\bar{X}] = \frac{1}{n}\sum_{i=1}^n E[X_i] = \frac{1}{n} \cdot n\mu = \mu$$

这里 $\mu$ 始终是个符号，从没有变成具体数字。这个推导对 $\mu = 3$ 成立，对 $\mu = 100$ 也成立，对任何 $\mu$ 都成立。

就好像你证明了``一把没有系统误差的尺子，量出来的平均值等于真实长度''——不需要知道真实长度是多少，这是尺子本身的结构性质。
\end{note}
x1，x2都是一次实验里面的个体
\begin{dxtips}
也就是说因为他服从这个分布所以不管选第一次实验x1，第二次实验x2都要把他的均值当作真实正态分布的均值对吗。而不是选取一次实验的最大值，最小值直接当他服从的正态分布的均值。而是选取他的平均值充当
\end{dxtips}
\begin{note}[UMVUE 解决的是选公式的问题]
你确实只做一次实验，得到一个数。但在做实验之前，你要选公式。

方差小的公式，代入数据后得到的数大概率更靠近真值；方差大的公式，这一次的结果可能差很远。

类比：你只射一箭，但要事先选弓。弓A精度高，弓B精度差。虽然只射一次，但选哪把弓决定了这一箭大概率落在哪里。UMVUE 就是在所有无偏的弓里，选精度最高的那把。

所以 UMVUE 是\textbf{选公式阶段}的理论，不是做完实验之后的事情。
\end{note}

\begin{example}[UMVUE 的具体比较]
设 $X_1, X_2, X_3, X_4 \overset{\text{i.i.d.}}{\sim} N(\mu, 1)$，估计 $\mu$。三个候选公式均无偏（期望均等于 $\mu$）：

$$\hat{\mu}_A = \bar{X} = \frac{X_1+X_2+X_3+X_4}{4}, \quad \mathrm{Var}(\hat{\mu}_A) = \frac{1}{4}$$

$$\hat{\mu}_B = \frac{X_1+X_2}{2}, \quad \mathrm{Var}(\hat{\mu}_B) = \frac{1}{2}$$

$$\hat{\mu}_C = X_1, \quad \mathrm{Var}(\hat{\mu}_C) = 1$$

方差衡量的是：这个公式在不同数据下的结果散布多广。$\bar{X}$ 的结果永远集中在 $\mu$ 附近，$X_1$ 的结果可能飘得很远。

在所有对 $\mu$ 的无偏估计量中，$\bar{X}$ 方差最小，故 $\bar{X}$ 是 $\mu$ 的 UMVUE。
\end{example}
\begin{note}
    这是统计里一个让人头疼的惯例，其实它们是两个不同的东西：

\begin{itemize}
  \item \textbf{实验前：$X_1$ 是随机变量（大写）}
  
  还不知道会观测到什么值，可能是 $160$，可能是 $180$，服从 $N(\mu, \sigma^2)$。这时候可以谈期望：$E[X_1] = \mu$。

  \item \textbf{实验后：$x_1$ 是观测值（小写）}
  
  已经量完了，是一个固定的数，比如 $172$。这时候不谈期望，直接代入算：$\bar{x} = (172 + 168 + \cdots)/n$。
\end{itemize}

因为教材里经常两个都写大写 $X_1$，没有严格区分。区分的方法就是看语境：出现 $E[\cdot]$ 就当随机变量（实验前）；出现具体数字代入就当观测值（实验后）。


\end{note}
\begin{dxtips}
    找无偏估计量永远发生在实验前，与具体采样永远无关。需要把所有可能出现的样本进行 $h(X)$ 的映射，再与其出现的概率做乘积，求和得到期望：
$$E[h(X)] = \sum_k h(k) \cdot P(X=k) = g(\theta)$$
这个期望等于我们想估计的目标 $g(\theta)$，即为无偏。
\end{dxtips}
\begin{note}
大写 $X_1$ 是``还没揭晓的牌''，小写 $x_1$ 是``已经翻开的牌''。期望只对没揭晓的牌讲。
\end{note}
\begin{dxtips}
    

所以无偏估计量是找一个公式，用这个公式代入所有可能的样本后，按照这些样本的分布分配给它们概率，然后这个期望等于我们想估计的东西。
\end{dxtips}

\subsection{fisher信息}
\begin{note}
\textbf{Fisher 想解决的问题：}这组数据对 $\theta$ 有多"敏感"？

如果 $\theta$ 稍微变一点，数据出现的概率就大幅变化——说明数据能精确分辨
$\theta$ 的位置，信息量大。如果怎么改 $\theta$ 似然函数都几乎不动，
说明数据对 $\theta$ 几乎无感，信息量小。
\end{note}

\begin{note}
得分函数 $S = \dfrac{\partial}{\partial\theta}\ln f(X;\theta)$ 正好衡量这件事：
它就是对数似然关于 $\theta$ 的导数，即似然函数对 $\theta$ 变化的敏感程度。
\begin{itemize}
  \item $S$ 大 $\to$ 似然对 $\theta$ 变化剧烈 $\to$ 数据信息多
  \item $S$ 小 $\to$ 似然对 $\theta$ 几乎无感 $\to$ 数据信息少
\end{itemize}
\end{note}
\begin{note}
$f(X;\theta)$ 是什么：是单个 $X$ 的概率密度函数

就是概率密度函数，但把视角反过来——数据 $X$ 已经观测到了，$\theta$ 是未知的。
$$f(X;\theta) = \text{``如果参数是 } \theta \text{，观测到这个 } X \text{ 的概率有多大？''}$$
这就是似然函数，$\theta$ 变，这个概率跟着变。
\begin{itemize}
  \item 单个 $X$：得分函数为 $S = \dfrac{\partial}{\partial\theta}\ln f(X;\theta)$，
        只用 $f(X;\theta)$，与 $L(\theta)$ 无关。
  \item 多个 $X_1,\dots,X_n$：似然函数 $L(\theta)=\prod_{i=1}^n f(X_i;\theta)$，
        得分函数为
        $$S_n = \frac{\partial}{\partial\theta}\ln L(\theta)
        = \sum_{i=1}^n \frac{\partial}{\partial\theta}\ln f(X_i;\theta)
        = \sum_{i=1}^n S(X_i;\theta)$$
        此时得分函数与 $L(\theta)$ 产生联系。
\end{itemize}
\end{note}
\begin{dxtips}
    ln是为了累乘变累加好算，对θ求导数因为本来就是腰算他的变化率
\end{dxtips}
\begin{note}
    似然函数英文叫 Likelihood function。
\textbf{Likelihood} 是"可能性/合理程度"的意思，不是严格的概率，
而是"这个 $\theta$ 值有多合理？"
中文"似然"就是翻译自 likelihood——似乎如此，好像这个参数值是对的。
\end{note}
\begin{itemize}
  \item $f(x \mid \theta)$：竖线后面是条件，贝叶斯里 $\theta$ 被当作随机变量，
        "在 $\theta$ 已知的条件下"
  \item $f(x;\theta)$：分号前面是\textbf{观测值}（已知），
        分号后面是\textbf{参数}（待估），频率派里 $\theta$ 不是随机变量，只是未知常数
\end{itemize}
\begin{note}
得分函数 $S(X;\theta)$ 本身是随机的，因为 $X$ 是随机的。

对于固定的 $\theta$，不同的 $X$ 值会给出不同的 $S$：
$$S(X;\theta) = \frac{\partial}{\partial\theta}\ln f(X;\theta)$$
$X$ 变，$S$ 就变。所以 $S$ 不是一个数，是一个随机变量。

我们已经知道 $E[S]=0$，均值没有信息。因为 $f(x;\theta)$ 是关于 $x$ 的概率密度，
积分恒为 $1$，两边对 $\theta$ 求导所以等于 $0$。

那用什么来总结"$S$ 平均能偏离 $0$ 多远"？自然就是方差：
\begin{itemize}
  \item $\operatorname{Var}(S)$ 大 $\to$ $S$ 经常取大值 $\to$ 似然对 $\theta$ 平均很敏感 $\to$ 信息多
  \item $\operatorname{Var}(S)$ 小 $\to$ $S$ 总在 $0$ 附近 $\to$ 似然对 $\theta$ 几乎无感 $\to$ 信息少
\end{itemize}
\end{note}
\begin{note}
    $E[S]=0$ 的推导依赖一步交换：
$$\frac{\partial}{\partial\theta}\int f(x;\theta)\,dx = \int \frac{\partial}{\partial\theta}f(x;\theta)\,dx$$
左边对 $\int f\,dx = 1$ 求导得 $0$，右边整理后就是 $E[S]$，所以 $E[S]=0$。

但这个交换要求\textbf{积分上下限不含 $\theta$}。$U(0,\theta)$ 的积分上限就是 $\theta$，交换不合法：
$$\frac{\partial}{\partial\theta}\int_0^\theta \frac{1}{\theta}\,dx = \frac{\partial}{\partial\theta}1 = 0$$
$$\int_0^\theta \frac{\partial}{\partial\theta}\frac{1}{\theta}\,dx = \int_0^\theta -\frac{1}{\theta^2}\,dx = -\frac{1}{\theta}$$
两边不相等，所以 $E[S] = -\dfrac{1}{\theta} \neq 0$，C-R 下界对 $U(0,\theta)$ 直接失效。
\end{note}
\subsection{CR}
\begin{note}
\textbf{C-R 下界在说什么：}

对任意无偏估计量 $\hat{\theta}$，方差不可能低于某个值：
$$\operatorname{Var}(\hat{\theta}) \geq \frac{1}{nI(\theta)}$$
就是给方差划了一条\textbf{不可逾越的底线}。
\end{note}

\begin{note}
\textbf{为什么被发明：}

能力二的核心问题是"哪个估计量更好"，判断标准是方差小。
但比较 A 和 B，你只知道 A 比 B 好，不知道 A 是不是最好的。

C-R 下界解决了这个问题——给你一个\textbf{绝对基准}：
\begin{itemize}
  \item $\operatorname{Var}(\hat{\theta}) = \dfrac{1}{nI(\theta)}$ $\to$ 最优，没有比它更好的无偏估计量
  \item $\operatorname{Var}(\hat{\theta}) > \dfrac{1}{nI(\theta)}$ $\to$ 还\textbf{可能}有改进空间
\end{itemize}
\end{note}
\newpage
\begin{proof}
    \textbf{第一步：}无偏条件 $E[\hat{\theta}] = \theta$，两边对 $\theta$ 求导：
$$\int \hat{\theta}\cdot\frac{\partial f}{\partial\theta}\,dx = 1$$
代入 $\textcolor{red}{\dfrac{\partial f}{\partial\theta} = f\cdot S}$，得：
$$E[\hat{\theta}\cdot S] = 1$$

\textbf{第二步：}因为 $E[S]=0$，所以：
$$\operatorname{Cov}(\hat{\theta},\, S) = E[\hat{\theta}\cdot S] - E[\hat{\theta}]\cdot E[S] = 1$$

\textbf{第三步：}柯西-施瓦茨不等式：
$$\operatorname{Cov}(\hat{\theta},S)^2 \leq \operatorname{Var}(\hat{\theta})\cdot\operatorname{Var}(S) \implies 1 \leq \operatorname{Var}(\hat{\theta})\cdot I(\theta) \implies \operatorname{Var}(\hat{\theta}) \geq \frac{1}{I(\theta)}$$
$n$ 个样本再除以 $n$ 得 $\dfrac{1}{nI(\theta)}$。
\end{proof}
\begin{dxtips}
柯西-施瓦茨不等式才能把无偏估计的方差连接到不等式中。
我们已经知道无偏估计的期望是 $\theta$，为了出现 $\dfrac{\partial f}{\partial\theta}$
从而引入 $S$，所以对两边求导，再利用 $E[S]=0$ 的信息
和 $\operatorname{Var}(S) = I(\theta)$ 完成这个证明。
\end{dxtips}
\begin{note}[正则条件]
正则条件就是允许积分和求导交换顺序的条件：
$$\frac{d}{d\theta}\int \hat{g}\cdot L(\theta;\boldsymbol{x})\,d\boldsymbol{x} = \int \hat{g}\cdot \frac{\partial L(\theta;\boldsymbol{x})}{\partial\theta}\,d\boldsymbol{x}$$
具体需要满足：
\begin{enumerate}
    \item 支撑集不依赖 $\theta$——积分范围不能随 $\theta$ 变
    \item $f(x;\theta)$ 对 $\theta$ 可微
    \item 求导后的被积函数有可积上界（控制收敛定理条件）
\end{enumerate}
\end{note}
\section{umvue}
\begin{note}
    \textbf{全称}：Uniformly Minimum Variance Unbiased Estimator，一致最小方差无偏估计量。

\textbf{为什么发明它}，从需求倒推：

你想估计 $\theta$，无偏是基本要求（平均不偏）。但满足无偏的估计量有无数个——$X_1$ 是无偏的，$\overline{X}$ 也是，随便凑个线性组合还是。\textbf{怎么选最好的那个？}

自然的标准：方差最小——离真值的随机误差最小。

但"最小"要小心：不能只对某个特定 $\theta$ 值最小，要对\textbf{所有} $\theta \in \Theta$ 同时最小，这就是"Uniformly"（一致）的含义。

所以 UMVUE 回答的问题是：
\begin{quote}
在全部无偏估计量里，有没有一个，不管真实参数取什么值，它的方差都比其他所有无偏估计量小？
\end{quote}
有的话，它就是 UMVUE，是无偏类里的理论最优。
\end{note}

    uniformity 一致的，一致收敛是uniform，m就是最小 v就是方差 ue就是无偏估计
\end{dxtips}
\subsection{完备充分统计量}
\begin{note}[充分统计量不能丢信息]
    用正态 $N(\mu, 1)$，$n=2$ 来说，取\textbf{不充分}的统计量 $T = X_1$。

你观测到 $X_1 = 3,\ X_2 = 5$。

\textbf{用 $T = X_1 = 3$ 去估计 $\mu$}：你只看第一个观测值，猜 $\mu \approx 3$。
但 $X_2 = 5$ 还摆在那里，它也服从 $N(\mu, 1)$，里面明显还有 $\mu$ 的信息——
看了 $X_2 = 5$ 之后你会更新判断，觉得 $\mu$ 应该在 4 附近。
$T = X_1$ 把 $X_2$ 里的 $\mu$ 信息\textbf{扔掉了}，这就是"丢了 $\theta$ 信息"。

\textbf{用 $T = \overline{X} = 4$ 就不丢}：
知道了 $\overline{X} = 4$，再去看 $X_1 = 3,\ X_2 = 5$ 的具体分法——
$X_1$ 比均值低 1，$X_2$ 比均值高 1——这件事和 $\mu$ 是多少完全没关系，只是随机波动。
没有任何新的 $\mu$ 信息可以被挖出来了。

也就是说，知道了 $\overline{X} = 4$，不用再知道到底是 $(3,5)$ 还是 $(4,4)$，
因为对"$\mu$ 大概是多少"这个问题，两种情况提供的信息量完全一样。

\textbf{一句话}：$T$ 是不是充分的，就看"知道 $T$ 之后，样本剩下的部分还能不能更新你对 $\theta$ 的判断"。能更新就说明丢了信息，不充分；不能更新就是充分的。
\end{note}
\begin{note}
\textbf{完备空间类比}

完备空间的核心是：极限不会"跑出去"——每个 Cauchy 列的极限都在空间里，没有"洞"。

$\mathbb{Q}$ 不完备：$1, 1.4, 1.41, 1.414, \ldots$ 是 Cauchy 列，极限 $\sqrt{2}$ 跑到 $\mathbb{Q}$ 外面去了。

$\mathbb{R}$ 完备：任何 Cauchy 列的极限都还在 $\mathbb{R}$ 里，无处可逃。

\textbf{完备充分统计量的逻辑完全一样：}

不完备的 $T$：存在非零函数 $g(T)$，它的期望"跑到零去了"（$E[g(T)]=0\ \forall\theta$），但 $g$ 本身不是零。
就像 $\sqrt{2}$ 逃出了 $\mathbb{Q}$——非零的东西期望上"假装是零"。

完备的 $T$：期望等于零这件事无处可逃，只要 $E[g(T)]=0$，$g$ 就必须是零本身，没有任何非零函数能"藏"在零期望后面。

\textbf{为什么这样定义}

设计这个定义的动机就是逼出唯一性。假设 $g_1(T)$ 和 $g_2(T)$ 都是 $\theta$ 的无偏估计，那么：
$$E_\theta[g_1(T) - g_2(T)] = 0 \quad \forall\theta$$
完备性直接给出 $g_1(T) = g_2(T)$——基于 $T$ 的无偏估计\textbf{唯一}，唯一的那个自然是 UMVUE。定义就是为了这一步设计的。
\end{note}
\begin{dxtips}
    所以完备统计量就抱着基于T的无偏估计只有一个，因为只有一个所以一定是umvue对吗
\end{dxtips}
\subsection{因子分解定理}
\begin{theorem}[Fisher-Neyman 因子分解定理]
$T(\boldsymbol{X})$ 是 $\theta$ 的充分统计量 $\iff$ 联合密度可分解为
$$f(\boldsymbol{x};\theta) = g(T(\boldsymbol{x}),\,\theta)\cdot h(\boldsymbol{x})$$
其中 $g$ 通过 $T$ 依赖数据，$h$ 完全不含 $\theta$。
\end{theorem}
\begin{note}[因子分解定理的直觉]
联合密度 $f(\boldsymbol{x};\theta) = \prod_{i=1}^n f(x_i;\theta)$ 包含了样本关于 $\theta$ 的全部信息。

因子分解定理说的是：若联合密度能写成
$$f(\boldsymbol{x};\theta) = g(T(\boldsymbol{x}),\,\theta)\cdot h(\boldsymbol{x})$$
则 $T(\boldsymbol{X})$ 是 $\theta$ 的充分统计量。

其中：
\begin{itemize}
    \item $g(T(\boldsymbol{x}),\theta)$：既含 $\theta$ 又含数据，但数据只通过 $T(\boldsymbol{x})$ 与 $\theta$ 互动。$T(\boldsymbol{x})$ 就是这个互动中数据那一侧的函数，即充分统计量。
    \item $h(\boldsymbol{x})$：只含数据，不含 $\theta$，与 $\theta$ 零互动——不管 $\theta$ 取何值，$h(\boldsymbol{x})$ 不变，对推断 $\theta$ 毫无贡献。
\end{itemize}
\end{note}
\begin{dxtips}
\textbf{两句话总结：}
\begin{itemize}
    \item 联合密度保证了关于 $\theta$ 的信息完整。
    \item 因子分解保证了 $T(\boldsymbol{X})$ 是与 $\theta$ 互动的全部，$h(\boldsymbol{x})$ 里没有任何关于 $\theta$ 的额外信息。
\end{itemize}
\end{dxtips}
\begin{example}[正态总体因子分解，$\sigma^2$ 已知]
$X_i \sim N(\mu,\sigma^2)$，联合密度展开后分离：
$$\prod_{i=1}^n f(x_i;\mu)
= \underbrace{\exp\!\left\{\frac{\mu\sum x_i}{\sigma^2} - \frac{n\mu^2}{2\sigma^2}\right\}}_{g(T,\,\mu),\quad T=\sum x_i}
\cdot
\underbrace{\left(\frac{1}{\sqrt{2\pi}\sigma}\right)^{\!n} \exp\!\left\{-\frac{\sum x_i^2}{2\sigma^2}\right\}}_{h(\boldsymbol{x}),\ \text{不含}\ \mu}$$
故 $T = \sum X_i$（即 $\bar{X}$）是 $\mu$ 的充分统计量。
\end{example}
\begin{note}[Rao-Blackwell：把任意无偏估计升级为基于充分统计量的无偏估计]
有了充分统计量 $T$，任何无偏估计 $\delta(\boldsymbol{X})$ 都可以改进：令
$$\varphi(T) = E[\delta(\boldsymbol{X}) \mid T]$$
则 $\varphi(T)$ 仍无偏，且方差不超过原来：$\mathrm{Var}(\varphi(T)) \leq \mathrm{Var}(\delta(\boldsymbol{X}))$。

\textbf{为什么要这么做：}$\delta(\boldsymbol{X})$ 可能只用了部分样本信息（如只看 $X_1$），
条件期望把其余样本的信息也并入进来，使 $\varphi(T)$ 等价于充分统计量，关于 $\theta$ 的信息更完整，方差因此下降。
\end{note}

\begin{dxtips}
$T$ 包含关于 $\theta$ 的全部信息，而 $\delta(\boldsymbol{X})$（如 $X_1$）可能只含部分信息。
计算条件期望 $E[\delta \mid T]$ 时，被迫引入其他样本的所有可能情况及其概率，
结果 $\varphi(T)$ 比原估计量含有更多关于 $\theta$ 的信息，方差因此下降。

一句话：Rao-Blackwell 不是``获得更多信息''，而是让估计量从``部分信息''升级到``充分统计量所含的全部信息''。
\end{dxtips}
\begin{note}[Lehmann-Scheffé 定理（勒曼-谢菲定理）]
若 $T$ 是\textbf{完备充分}统计量，$\varphi(T)$ 是 $g(\theta)$ 的无偏估计，则 $\varphi(T)$ 是 $g(\theta)$ 的 UMVUE，且唯一。

\textbf{为什么完备性保证了唯一最优：}

假设有两个基于 $T$ 的无偏估计 $\varphi_1(T)$ 和 $\varphi_2(T)$，则
$$E[\varphi_1(T) - \varphi_2(T)] = 0 \quad \forall\theta$$
完备性说：期望恒为零 $\Rightarrow$ 函数本身就是零，即 $\varphi_1(T) = \varphi_2(T)$ a.s.

基于完备充分统计量的无偏估计\textbf{只有一个}，唯一的那个自然是所有无偏估计里方差最小的。
\end{note}
\subsection{指数族的优越性}
\begin{note}[指数族的常见成员]
\begin{itemize}
    \item 正态分布 $N(\mu,\sigma^2)$
    \item 泊松分布 $P(\lambda)$
    \item 二项分布 $B(n,p)$，$n$ 已知
    \item 指数分布 $\mathrm{Exp}(\lambda)$
    \item Gamma 分布 $\Gamma(\alpha,\beta)$
    \item 伯努利分布 $B(1,p)$
\end{itemize}
\textbf{不是指数族：}均匀分布 $U(0,\theta)$（支撑集依赖 $\theta$，无法写成标准形式）。
\end{note}
\begin{note}[指数族标准形式：泊松与正态对比]
标准形式：$f(x;\theta) = h(x)\cdot\exp\{Q(\theta)\cdot t(x) - A(\theta)\}$

\textbf{泊松 $P(\lambda)$：}
$$f(x;\lambda) = \frac{\lambda^x e^{-\lambda}}{x!}
= \underbrace{\frac{1}{x!}}_{h(x)}
\cdot\exp\!\left\{
\underbrace{\ln\lambda}_{Q(\lambda)}\cdot
\underbrace{x}_{t(x)}-
\underbrace{\lambda}_{A(\lambda)}
\right\}$$

\textbf{正态 $N(\mu,\sigma^2)$，$\sigma^2$ 已知：}
$$f(x;\mu)
= \underbrace{\frac{1}{\sqrt{2\pi}\sigma}e^{-\frac{x^2}{2\sigma^2}}}_{h(x)}
\cdot\exp\!\left\{
\underbrace{\frac{\mu}{\sigma^2}}_{Q(\mu)}\cdot
\underbrace{x}_{t(x)}-
\underbrace{\frac{\mu^2}{2\sigma^2}}_{A(\mu)}
\right\}$$
\end{note}
\begin{note}[为什么指数族的充分统计量是完备的]
指数族密度含 $e^{Q(\theta)\cdot t(x)}$，充分统计量 $T=\sum t(X_i)$ 的密度为
$$f_T(s;\theta) = \tilde{h}(s)\cdot e^{Q(\theta)\cdot s}$$
证完备性时，假设 $E_\theta[\varphi(T)]=0$，展开得
$$\int \varphi(s)\cdot\tilde{h}(s)\cdot e^{Q(\theta)\cdot s}\,ds = 0 \quad \forall\theta$$
令 $\eta=Q(\theta)$，这正是函数 $\varphi(s)\tilde{h}(s)$ 的\textbf{拉普拉斯变换恒为零}。

由拉普拉斯变换的唯一性：变换恒为零 $\Rightarrow$ 原函数为零，即 $\varphi(T)=0$ a.s.

指数族的 $e^{Q(\theta)\cdot s}$ 结构天然就是拉普拉斯变换的形式，完备性由此而来。
\end{note}
\begin{dxtips}
    看见指数分布族只要无偏+完备统计量他就是umvue
\end{dxtips}
\section{相和与渐进正态}
\subsection{相和}
\begin{note}
    $g(\theta)$ 是为了写得更一般——它在估计 $\theta$ 的某个函数，不一定是 $\theta$ 本身。

比如：
\begin{itemize}
    \item 想估计 $\theta$，那 $g(\theta)=\theta$，$\hat{g}_n \to \theta$
    \item 想估计 $\theta^2$，那 $g(\theta)=\theta^2$，$\hat{g}_n \to \theta^2$
    \item 想估计 $e^\theta$，那 $g(\theta)=e^\theta$，$\hat{g}_n \to e^\theta$
\end{itemize}
\end{note}
\begin{dxtips}
   \begin{enumerate}
       \item 简单来说，$g(\theta)$ 就是我们要估计的量，写成这种形式就代表它是和未知参数 $\theta$ 相关的。\item $\hat{g}_n(X_1, X_2, \cdots, X_n)$ 是一个函数，代入 $n$ 个数据，
输出的结果就是我们对 $g(\theta)$ 的估计值。
   \end{enumerate}
\end{dxtips}
\begin{note}
    相合性定义的是一个宏观标准：随着数据增多，
如果 $\hat{g}_n$ 这套公式的输出能收敛到真实值 $g(\theta)$，
我们就认为它是一个好的估计公式。

反过来，如果数据堆再多结果也收敛不过去，这套公式就不值得用。
\end{note}
\begin{dxtips}
    强想和是几乎处处收敛，就是不收敛的是一个固定的零测集，弱收敛依概率收敛只要求每个固定 $\varepsilon$ 下``超出误差''的概率趋于零，
但那个坏集合可以到处飘；
几乎处处收敛要求整条路径几乎必然最终落在真值上，更强。
\end{dxtips}
\begin{note}
    强想和经过线形变化还是强想和，也就强想和的公式经过线性变化还是好公式
\end{note}
\begin{note}[均方误差（MSE）]
MSE（Mean Squared Error，均方误差）度量估计量和真值之间的平均平方距离，
可以拆成两项：
$$
E\!\left[(\hat{g}_n - g(\theta))^2\right]
= \underbrace{\left[E\hat{g}_n - g(\theta)\right]^2}_{\text{偏差}^2}
+ \underbrace{\mathrm{Var}(\hat{g}_n)}_{\text{方差}}
$$
其中\textbf{偏差} $= E[\hat{g}_n] - g(\theta)$，衡量估计量的期望和真值之间的系统性差距。

两项的直觉：
\begin{itemize}
    \item 偏差：瞄准方向本身就歪了（系统误差）
    \item 方差：即使方向对，每次结果也在抖动（随机误差）
\end{itemize}
\end{note}
\begin{dxtips}
    方差的定义是 $E[(\hat{g}_n - E\hat{g}_n)^2]$，所以在括号里加减 $E\hat{g}_n$：
$$(\hat{g}_n - \theta^*) = \underbrace{(\hat{g}_n - E\hat{g}_n)}_{\text{方差来源}} + \underbrace{(E\hat{g}_n - \theta^*)}_{\text{偏差来源}}$$
平方展开后出现一次交叉项，但由于 $E[\hat{g}_n - E\hat{g}_n] = 0$ 恒成立，交叉项消掉，
只剩方差和偏差$^2$两项。
\end{dxtips}
\begin{note}
    \textbf{均方收敛（MSE $\to$ 0）}

核心结论：若 MSE 趋于 0，则相合。
$$\mathrm{MSE}(\hat{g}_n) = E[(\hat{g}_n - g(\theta))^2] \to 0 \implies \hat{g}_n \xrightarrow{P} g(\theta)$$

\textbf{为什么？} 切比雪夫不等式：
$$P(|\hat{g}_n - g(\theta)| > \varepsilon) \leq \frac{\mathrm{MSE}(\hat{g}_n)}{\varepsilon^2} \to 0$$

\textbf{MSE 拆开来是两项：}
$$\mathrm{MSE} = \text{偏差}^2 + \text{方差} = [E\hat{g}_n - g(\theta)]^2 + \mathrm{Var}(\hat{g}_n)$$

所以要证 MSE $\to 0$，只需证这两项\textbf{都趋于 0}：
\begin{itemize}
    \item 偏差$^2 \to 0$（估计量越来越不偏）
    \item 方差 $\to 0$（估计量越来越集中）
\end{itemize}

\textbf{实际用法：}无偏估计量偏差 $= 0$，只需证方差 $\to 0$ 即可。这是最常用的情形。
\end{note}
\begin{note}
    \textbf{大数定律}：样本均值收敛到总体期望
$$\bar{X}_n = \frac{1}{n}\sum_{i=1}^n X_i \xrightarrow{P} E[X]$$
更一般地，任何样本矩都收敛到对应的总体矩。

\textbf{Slutsky 定理}：收敛可以穿过连续函数
$$T_n \xrightarrow{P} c \implies g(T_n) \xrightarrow{P} g(c) \qquad (g \text{ 在 } c \text{ 处连续})$$
\end{note}
\begin{note}[验证相合性：Slutsky + 大数定律]
不算任何期望和方差，只看结构——
\begin{center}
\textit{这个估计量是不是样本矩的某个连续函数？}
\end{center}
如果是，大数定律直接给样本矩的收敛，Slutsky 直接传递过去，一步到位。

矩估计量天生就是这个结构，所以对矩估计量用这条路根本不需要展开 MSE。

两条路结论一样，只是省不省计算而已。
\end{note}
\begin{dxtips}
    \textbf{第一步，大数定律}建立收敛：
$$\bar{X}_n = \frac{1}{n}\sum_{i=1}^n X_i \xrightarrow{P} E[X]$$
即样本均值依概率收敛到总体期望，提供 $T_n \xrightarrow{P} c$ 这个条件。

\textbf{第二步，连续映射 定理}传递收敛：
$$T_n \xrightarrow{P} c \implies g(T_n) \xrightarrow{P} g(c) \qquad (g \text{ 在 } c \text{ 处连续})$$

对 $k$ 阶样本矩同理：$\dfrac{1}{n}\sum_{i=1}^n X_i^k \xrightarrow{P} E[X^k]$，连续映射定理 照常传递。
\end{dxtips}
\begin{note}
    \textbf{Slutsky 定理完整表述：}

若 $X_n \xrightarrow{d} X$，$Y_n \xrightarrow{P} c$（常数），则：
\begin{enumerate}
    \item $X_n + Y_n \xrightarrow{d} X + c$
    \item $X_n \cdot Y_n \xrightarrow{d} c \cdot X$
    \item $X_n / Y_n \xrightarrow{d} X / c \quad (c \neq 0)$
\end{enumerate}

关键有两点是``连续不打断收敛''没说清楚的：

\textbf{第一}：必须一个依分布、一个依概率收敛到\textbf{常数}，两个都依分布不能用。

\textbf{第二}：$Y_n$ 收敛的目标必须是\textbf{常数}，不能是随机变量。

这也是这道题先把 $Y_n \xrightarrow{a.s.} a$ 转成 $\dfrac{1}{Y_n} \xrightarrow{P} \dfrac{1}{a}$ 的原因——
要凑出``依概率收敛到常数''这个形式，Slutsky 才能用。
\end{note}
\begin{dxtips}
    要用Slutsky定理就得保证一个依分布收敛也就是d- distribution 另一个第一必须是依概率收敛第二必须依概率收敛到常数
\end{dxtips}
\begin{note}
    \begin{itemize}
    \item \textbf{依分布收敛}（$X_n \xrightarrow{d} X$）：保证了结果的极限分布形状是 $X$，提供``收敛到谁''这个内容。
    \item \textbf{依概率收敛}（$Y_n \xrightarrow{P} c$）：保证了 $Y_n$ 的取值高概率集中在 $c$ 附近，误差 $|Y_n - c|$ 可以被控制住，使得 $X_n \cdot Y_n \approx X_n \cdot c$ 这个近似在极限下成立。
    \item \textbf{收敛到常数}（$c$ 是常数而非随机变量）：消掉了联合分布的问题。常数没有随机性，不需要知道 $X_n$ 和 $Y_n$ 之间的相关结构，两个变量的联合行为自动退化成可操控的形式。
\end{itemize}
\end{note}
\section{MLE与渐近正态分布}
\begin{note}[MLE就是一种构造估计量的方式]

    \textbf{第一步：写似然函数}

数据 $X_1, \ldots, X_n$ 独立同分布，联合概率为：
$$L(\theta) = \prod_{i=1}^n f(X_i, \theta)$$
$L(\theta)$ 的含义：在参数为 $\theta$ 时，观测到这组数据的概率。

\textbf{第二步：取对数}

$$\ell(\theta) = \ln L(\theta) = \sum_{i=1}^n \ln f(X_i, \theta)$$
取对数不改变最大值位置，但将连乘化为求和，便于求导。

\textbf{第三步：求最大值}

$$\ell'(\theta) = 0 \quad \Longrightarrow \quad \hat{\theta}_{\mathrm{MLE}}$$

\textbf{核心思想：}数据已经发生，反推使这组数据
"最可能出现"的 $\theta$，即为 MLE。
\end{note}
\begin{dxtips}
   \textbf{Maximum Likelihood Estimation 理解}

\begin{itemize}
  \item \textbf{Max}：要对似然函数求极大值，即求导令导数为零。
  \item \textbf{Likelihood}：似然函数 $L(\theta) = \prod_{i=1}^n f(X_i, \theta)$，
        即在参数为 $\theta$ 时观测到这组数据的概率。
  \item \textbf{Estimation}：解出 $\hat{\theta}$ 作为估计量。
\end{itemize}

为方便求导，先取对数：
$$\ell(\theta) = \ln L(\theta) = \sum_{i=1}^n \ln f(X_i, \theta)$$
将连乘化为求和，不改变极值位置，再令 $\ell'(\theta) = 0$ 解出 $\hat{\theta}$。

\textbf{数学意义：}在现实已经发生情况 $A$ 的条件下，
是哪个 $\theta$ 导致 $A$ 发生的可能性最大。

\textbf{小补充：}$\theta$ 是固定未知常数，非随机变量，
不能说"$\theta$ 是某个值的概率"。
MLE 求的是 $L(\theta)=P(\text{数据}\mid\theta)$ 对 $\theta$ 的最大，
而非贝叶斯的 $P(\theta\mid\text{数据})$。
\end{dxtips}
\begin{note}
    MLE 解出来的 $\hat{\theta}$ 是一个数，但有两个问题：

\begin{itemize}
  \item 问题1：不知道它有没有偏（有限样本下 MLE 通常有偏）
  \item 问题2：即使知道有偏，也不知道误差长什么形状
\end{itemize}

所以光有 $\hat{\theta}$ 这个数是不够的，还需要知道它的分布。
渐近正态就是答案：
$$\hat{\theta} \approx N\!\left(\theta,\, \frac{1}{nI(\theta)}\right) \quad (n \text{ 很大时})$$
有了这个分布就知道了：
\begin{itemize}
  \item 误差有多大（渐近方差 $\frac{1}{nI(\theta)}$）
  \item 误差是正态形状（可以做区间估计、假设检验）
  \item 渐近方差恰好等于 C-R 下界（MLE 在大样本下是最优的）
\end{itemize}

MLE 给你一个点 $\hat{\theta}$，渐近正态告诉你这个点周围的误差分布。

MLE 算出确定值但不知道好不好，渐近正态就是用来回答「好不好」这个问题的。
\end{note}
\begin{dxtips}
    MLE就是算得分函数什么时候是0
\end{dxtips}
\subsection{渐近正态}
\begin{definition}[渐近正态性]
设 $\hat{\theta}_n$ 是 $\theta$ 的估计量，若存在 $\sigma^2(\theta)$ 使得：
$$\sqrt{n}(\hat{\theta}_n - \theta) \xrightarrow{d} N(0,\, \sigma^2(\theta))$$
则称 $\hat{\theta}_n$ 是 $\theta$ 的\textbf{渐近正态估计量}，
$\sigma^2(\theta)$ 称为\textbf{渐近方差}，等价地：
$$\hat{\theta}_n \xrightarrow{d} N\!\left(\theta,\, \frac{\sigma^2(\theta)}{n}\right)$$
对 MLE，$\sigma^2(\theta) = \dfrac{1}{I(\theta)}$，故：
$$\hat{\theta} \xrightarrow{d} N\!\left(\theta,\, \frac{1}{nI(\theta)}\right)$$
\end{definition}
\begin{dxtips}
渐近存在的意义就是说MLE这种近似是好的
    ，而且渐近正态不只说"好"，而是说好在哪里、好多少：
\begin{itemize}
  \item 好在哪里：渐近无偏（中心落在真实 $\theta$）
  \item 好多少：渐近方差 $= \dfrac{1}{nI(\theta)} =$ C-R 下界
        （没有任何估计量能比它更好）
\end{itemize}
渐近正态证明的是：MLE 在大样本下是最优无偏估计量，这叫\textbf{渐近有效}。
\end{dxtips}

\begin{note}
    MLE 渐近正态的证明核心是对 $\ell'(\hat{\theta})=0$ 在真实 $\theta$ 处 Taylor 展开：
$$0 = \ell'(\hat{\theta}) \approx \ell'(\theta) + (\hat{\theta} - \theta)\ell''(\theta)$$
整理得：
$$\sqrt{n}(\hat{\theta} - \theta) \approx \frac{\frac{1}{\sqrt{n}}\ell'(\theta)}{-\frac{1}{n}\ell''(\theta)}$$

\begin{itemize}
  \item 分子：CLT $\Rightarrow N(0,\, I(\theta))$ \quad $\leftarrow$ Fisher 从这里进来
  \item 分母：LLN $\Rightarrow I(\theta)$ \quad $\leftarrow$ Fisher 又出现一次
  \item 结果：Slutsky $\Rightarrow N\!\left(0,\, \dfrac{1}{I(\theta)}\right)$
\end{itemize}


\end{note}
\begin{dxtips}
    移项以后两边乘根号n,右侧上下同时除以n，得分函数期望是0，所以上面有了1/根号n就能用中心极限定理凑出一个依分布收敛到正态分布，下面由于有1/n还是独立同分布就可以说他依概率收敛到一个具体的数字也就是fisher,然后一个依分布一个依概率就可以用Slutsky了。最后方差除以常数要把常数平方
    
\end{dxtips}
\begin{dxtips}
    Fisher 信息有两个等价定义：
$$I(\theta) = E\!\left[\left(\frac{\partial \ln f}{\partial\theta}\right)^2\right] = -E\!\left[\frac{\partial^2 \ln f}{\partial\theta^2}\right]$$
左边是得分函数的方差（均值为 $0$，故方差 $=$ 二阶矩），
右边是二阶导的负期望，两个定义相等。
\end{dxtips}
\begin{note}
    \begin{itemize}
  \item \textbf{CLT（中心极限定理）：}$n$ 个 iid 随机变量，
        均值 $\mu$，方差 $\sigma^2$，则
        $$\frac{1}{\sqrt{n}}\sum_{i=1}^n (X_i - \mu) \xrightarrow{d} N(0,\sigma^2)$$
        不管原来什么分布，加起来除以 $\sqrt{n}$，极限就是正态。

  \item \textbf{LLN（大数定律）：}$n$ 个 iid 随机变量，均值 $\mu$，则
        $$\frac{1}{n}\sum_{i=1}^n X_i \xrightarrow{p} \mu$$
        样本均值收敛到总体期望。
\end{itemize}
\end{note}
\begin{theorem}[Delta 方法]
设 $\sqrt{n}(\hat{\theta} - \theta) \xrightarrow{d} N(0, \sigma^2)$，
$g$ 在 $\theta$ 处可导且 $g'(\theta) \neq 0$，则：
$$\sqrt{n}(g(\hat{\theta}) - g(\theta)) \xrightarrow{d} N(0,\, [g'(\theta)]^2 \sigma^2)$$
\end{theorem}
 \begin{note}[Delta 方法]
\textbf{核心思路：用线性化"转移"分布}。已知 $\sqrt{n}(\bar{X} - \lambda) \xrightarrow{d} N(0, \lambda)$，想知道 $\sqrt{n}(\bar{X}^2 - \lambda^2)$ 的极限分布。直接算不行，但注意到 $g(x) = x^2$ 在 $\lambda$ 附近是\textbf{光滑的}，可以 Taylor 展开：
$$g(\bar{X}) = g(\lambda) + g'(\lambda)(\bar{X} - \lambda) + \underbrace{O((\bar{X} - \lambda)^2)}_{\text{高阶项}}$$
所以 $g(\bar{X}) - g(\lambda) \approx g'(\lambda) \cdot (\bar{X} - \lambda)$，两边同乘 $\sqrt{n}$：
$$\sqrt{n}\bigl(g(\bar{X}) - g(\lambda)\bigr) \approx g'(\lambda) \cdot \sqrt{n}(\bar{X} - \lambda)$$
右边是已知分布乘以一个\textbf{常数}，这就能算了。
\end{note}
\begin{dxtips}
   因为 $g'(\lambda)$ 是一个常数，所以乘进去方程的时候需要平方。
$$\sqrt{n}(g(\hat{\theta}) - g(\theta)) \approx g'(\theta) \cdot \sqrt{n}(\hat{\theta} - \theta) \xrightarrow{d} N(0,\, [g'(\theta)]^2\sigma^2)$$
\end{dxtips}
\begin{note}[中心极限定理详解]
 Central Limit Theorem
\[
\text{Var}\!\left(\sum X_i - n\mu\right) = \text{Var}\!\left(\sum X_i\right) = \sum_{i=1}^n \text{Var}(X_i) = n\sigma^2
\]

% 常数不影响方差；独立 → 方差可以直接加；n 个 σ² 加起来 = nσ²

\text{三种情况：}
\begin{itemize}
  \item 直接看和：$\displaystyle\sum X_i - n\mu$，方差 $= n\sigma^2 \to \infty$（爆炸）
  \item 除以 $n$（样本均值）：$\bar{X} - \mu$，方差 $= \dfrac{\sigma^2}{n} \to 0$（塌缩成一个点，LLN）
  \item 除以 $\sqrt{n}$：$\dfrac{1}{\sqrt{n}}\displaystyle\sum(X_i-\mu)$，方差 $= \sigma^2$（稳定！）
\end{itemize}

$\sqrt{n}$ 是让方差既不爆炸也不消失的唯一选择。
\end{note}
\begin{dxtips}
因为 $n$ 个方差之和才是 $n\sigma^2$，
有的直接拿的是 $\bar{X}$，相当于已经除以了一个 $n$，
除多了，所以需要乘 $\sqrt{n}$ 恢复——
相当于原来乘 $\frac{1}{\sqrt{n}}$。

$$\text{Var}\!\left(\sqrt{n}\,(\bar{X}-\mu)\right)
= n \cdot \text{Var}(\bar{X})
= n \cdot \frac{\sigma^2}{n} = \sigma^2$$
\end{dxtips}

\section{置信区间}
\begin{dxtips}
具体一次含有 $\theta$ 就是含有，不含有就是不含有，就像一个具体的人身高是 180 就是一米八。只不过我们构造的置信区间是一个选择人群的方法，我们保证的是这个人群里面有 $1-\alpha$ 的可能性抽出一个 180。
\end{dxtips}
\begin{note}
置信区间是一套\textbf{构造区间的公式}，不是一个固定的区间。类比无偏估计量：

\begin{center}
\begin{tabular}{lll}
无偏估计量 & $\bar{X} = \dfrac{1}{n}\sum X_i$ & 公式，代入数据得一个\textbf{点} \\[10pt]
置信区间   & $\left[\bar{X} - z_{\alpha/2}\dfrac{\sigma}{\sqrt{n}},\ \bar{X} + z_{\alpha/2}\dfrac{\sigma}{\sqrt{n}}\right]$ & 公式，代入数据得一个\textbf{区间}
\end{tabular}
\end{center}

两者都是\textbf{实验前选好的公式}，实验后代入具体样本才变成具体的数。唯一的区别是：
\begin{itemize}
  \item 无偏估计量评价的是「这个公式平均不偏」
  \item 置信区间评价的是「这个公式长期套住 $\theta$ 的频率 $\geq 1-\alpha$」
\end{itemize}
\end{note}

\begin{dxtips}
置信度 $1-\alpha$ 是\textbf{公式的性质}，不是某次具体区间的性质——这就是为什么不能说「$\mu$ 有 $95\%$ 的概率在这个区间里」，只能说「这套公式有 $95\%$ 的靠谱率」。
\end{dxtips}
\begin{note}
置信区间的结构是：
$$\underbrace{\bar{X}}_{\text{中心}} \pm \underbrace{z_{\alpha/2}\dfrac{\sigma}{\sqrt{n}}}_{\text{误差限（半宽）}}$$

其中 $z_{\alpha/2}$ 是标准正态分布 $N(0,1)$ 的上 $\alpha/2$ 分位数，满足
$P(Z > z_{\alpha/2}) = \alpha/2$，是一个查表得到的常数。

误差限拆开看：
\begin{itemize}
  \item $z_{\alpha/2}$：由置信度决定，置信度越高，$z_{\alpha/2}$ 越大
  \item $\sigma$：总体本身的波动，越散误差限越大
  \item $\sqrt{n}$：样本量越大误差限越小，区间越窄
\end{itemize}
\end{note}

\begin{dxtips}
想要更窄的区间，本质上就是在压缩误差限：要么降置信度（$z_{\alpha/2}$ 变小），要么增大样本量（$\sqrt{n}$ 变大）。
\end{dxtips}
\begin{note}[正确理解]
想象你做了 100 次实验，每次抽一批样本，每次用同样的方法构造一个置信区间：

\begin{center}
\begin{tabular}{lllll}
第1次样本 & $\to$ & 区间 $[168.1,\ 172.5]$ & $\leftarrow$ & 包含了真实 $\mu$ \\
第2次样本 & $\to$ & 区间 $[169.0,\ 173.4]$ & $\leftarrow$ & 包含了真实 $\mu$ \\
第3次样本 & $\to$ & 区间 $[171.8,\ 176.2]$ & $\leftarrow$ & 没包含（倒霉）\\
$\vdots$  &       & $\vdots$               &              & \\
第100次   & $\to$ & $\cdots$               &              &
\end{tabular}
\end{center}

$95\%$ 置信度的意思是：这 100 个区间里，平均有 95 个会把真实的 $\mu$ 套住。

具体到某一次抽样得到的那个区间，你不知道它是不是那幸运的 95 个之一——但你知道这套方法长期来看靠谱率是 $95\%$。
\end{note}

\begin{note}[可靠度和精度的取舍]
想要 $99\%$ 置信度？可以，但区间会更宽（更不精确）。

想要窄区间？只能降低置信度，或者增大样本量。
\end{note}

\begin{dxtips}
越可靠，区间越长；区间越长，精度越低。
\end{dxtips}
\begin{note}[单侧与双侧置信区间]
\textbf{双侧}是在问：$\theta$ 在哪里？（两边都要套住）

\textbf{单侧}是在问：$\theta$ 有没有超过某个危险线？（只关心一边）

\textbf{双侧区间}——两边都套：
$$P(\theta_1(\boldsymbol{X}) \leq \theta \leq \theta_2(\boldsymbol{X})) \geq 1-\alpha$$

\textbf{置信上限}——只关心「$\theta$ 最多能有多大」：
$$P(\theta \leq \theta_2(\boldsymbol{X})) \geq 1-\alpha$$

\textbf{置信下限}——只关心「$\theta$ 至少有多大」：
$$P(\theta \geq \theta_1(\boldsymbol{X})) \geq 1-\alpha$$

单侧的「网」是开口的：
\begin{center}
\begin{tabular}{ll}
双侧 & $[\text{————}\theta\text{————}]$\quad 两边都围住，问「$\theta$ 在哪」\\[4pt]
置信上限 & $(\text{——————————}\theta]$\quad 只管右边，问「$\theta$ 会不会太大」\\[4pt]
置信下限 & $[\theta\text{——————————})$\quad 只管左边，问「$\theta$ 会不会太小」
\end{tabular}
\end{center}

单侧和双侧的分位数不同：置信度相同时，双侧两边各切 $\alpha/2$，查 $z_{\alpha/2}$；单侧只切一边，查 $z_\alpha$。因此单侧区间比双侧更窄——全部误差预算集中在一侧。
\end{note}

\begin{dxtips}
\[
\underbrace{[\,\theta_1(\boldsymbol{X}),\ \theta_2(\boldsymbol{X})\,]}_{\text{抽象，适用任何分布}}
\xrightarrow{\text{正态总体}(\sigma^2\text{已知），枢轴量法}}
\underbrace{\left[\,\bar{X} - z_{\alpha/2}\frac{\sigma}{\sqrt{n}},\ \bar{X} + z_{\alpha/2}\frac{\sigma}{\sqrt{n}}\,\right]}_{\theta_1 \text{ 和 } \theta_2 \text{ 的具体长相}}
\]
\end{dxtips}
\begin{itemize}
  \item $\boldsymbol{X} = (X_1, X_2, \ldots, X_n)$：整批样本，你抽到的所有数据
  \item $\theta_1(\boldsymbol{X})$：区间左端点，样本的函数，代入数据得一个数
  \item $\theta_2(\boldsymbol{X})$：区间右端点，同上
\end{itemize}
\section{枢轴量法（Pivotal quantity method）}

    \begin{itemize}

\item 原来我们要求的是 $P(f(X) < \theta < p(X)) = 1-\alpha$，原来我们是想知道这个 $f(X)$ 和 $p(X)$ 这两个函数是什么，但是我们压根不知道，因为中间的 $\theta$ 我们也不知道。

\item 于是呢，我们需要对 $\theta$ 进行一点变换，使得中心这个位置——这个位置其实就是核心，是枢纽，也就是 pivot——我们知道。于是我们就把原本的 $\theta$ 变成了 $\dfrac{\bar{X} - \theta}{\sigma / \sqrt{n}}$，因为这样 pivot 这整个的分布我们就知道了。

\item $\theta$ 从原来必须知道是多少，现在它已经完成了将正态从以 $\theta$ 为中心移动到以 $0$ 为核心的使命。

\item 然后为了等式成立，我们知道了两侧是 $z_{\alpha/2}$ 和 $-z_{\alpha/2}$，但是我们要求的其实是 $\theta$ 两边的函数是什么，所以把枢纽位置还原，还原的过程两边函数也得跟着变，这样两边的函数就知道是什么了。$\theta$必须是单调的才能反转

\end{itemize}
\begin{dxtips}
  \begin{itemize}
    \item 其实就是保证中心的枢纽改变以后我们知道它的分布。比如 $\mu$ 未知，$\sigma^2$ 也不知道，就没法用 $N(0,1)$ 了，因为不知道 $\sigma^2$，所以没法把方差弄到 $1$，只能用实际的方差，然后弄成 $t$ 分布了。
    \item 构造完了以后，里面只能留要求的量是未知的，其他的都得是已知的。
\end{itemize}
\end{dxtips}
\textbf{正态总体的标准情形：}
\begin{enumerate}
    \item \textbf{求 $\mu$}，$\sigma^2$ 已知 $\rightarrow$ 用 $Z$ 统计量
    $$Z = \frac{\bar{X} - \mu}{\sigma/\sqrt{n}} \sim N(0,1)$$

    \item \textbf{求 $\mu$}，$\sigma^2$ 未知 $\rightarrow$ 用 $t$ 统计量
    $$T = \frac{\bar{X} - \mu}{S/\sqrt{n}} \sim t(n-1)$$

    \item \textbf{求 $\sigma^2$}，$\mu$ 已知 $\rightarrow$ 用 $\chi^2$ 统计量
    $$\chi^2 = \frac{\sum_{i=1}^n(X_i-\mu)^2}{\sigma^2} \sim \chi^2(n)$$

    \item \textbf{求 $\sigma^2$}，$\mu$ 未知 $\rightarrow$ 用 $\chi^2$ 统计量
    $$\chi^2 = \frac{(n-1)S^2}{\sigma^2} \sim \chi^2(n-1)$$

    \item \textbf{两总体，求 $\mu_1-\mu_2$}（$\sigma_1^2=\sigma_2^2$ 未知）$\rightarrow$ 用 $t$ 统计量
    $$T = \frac{(\bar{X}-\bar{Y})-(\mu_1-\mu_2)}{S_p\sqrt{1/m+1/n}} \sim t(m+n-2)$$

    \item \textbf{两总体，求 $\sigma_1^2/\sigma_2^2$}，$\mu_1,\mu_2$ 未知 $\rightarrow$ 用 $F$ 统计量
    $$F = \frac{S_1^2/\sigma_1^2}{S_2^2/\sigma_2^2} \sim F(m-1,\,n-1)$$
\end{enumerate}
\begin{dxtips}
$\mu_1 - \mu_2$ 就是变相的 2，5 就是变相的 3。

F 分布就是两个独立卡方分布之比。

定义：若 $U \sim \chi^2(m)$，$V \sim \chi^2(n)$，且相互独立，则
$$F = \frac{U/m}{V/n} \sim F(m, n)$$
\end{dxtips}
\begin{dxtips}
   两样本 $t$ 统计量不是两个 $t$ 的叠加，而是将差 $\bar{X}-\bar{Y}$ 整体作为分子，用合并方差
$$S_p^2 = \frac{(m-1)S_1^2+(n-1)S_2^2}{m+n-2}$$
作为分母，结构上与单样本 $t$ 完全一致：
$$T = \frac{(\bar{X}-\bar{Y})-(\mu_1-\mu_2)}{S_p\sqrt{1/m+1/n}} \sim t(m+n-2)$$
\end{dxtips}
 \textbf{大样本置信区间}
\begin{note}
   

使用场景：总体分布未知，但 $n$ 足够大。

核心就是 CLT：无论总体是什么分布，只要 $n$ 大，
$$\frac{\bar{X} - \mu}{\sigma/\sqrt{n}} \xrightarrow{d} N(0,1)$$
$\sigma$ 未知也没关系，用样本标准差 $S$ 替换，近似仍成立：
$$Z = \frac{\bar{X} - \mu}{S/\sqrt{n}} \approx N(0,1)$$
也就是说，只要样本够大，$\sigma^2$ 不知道也可以用 $Z$ 而不是 $T$。
\end{note}

    \item \textbf{指数分布，求 $\theta$}，$X_1 \sim \dfrac{1}{\theta}e^{-x/\theta}$ \\

    $$X_{(1)}-b \leq \theta \leq X_{(1)}-a$$
\end{enumerate}

\dirtree{%
.1 假设检验框架.
.2 两种现实 $\times$ 两种决定.
.3 $H_0$ 为真，拒绝 $H_0$ $\;\to\;$ 第一类错误（弃真），概率 $= \alpha$.

.3 $\theta \in \Theta_1$：$1 - \beta(\theta) = \beta$（第二类错误）.
.2 $p$ 值：$H_0$ 成立时，观测到当前统计量或更极端结果的概率.
.3 $p < \alpha$ $\;\to\;$ 拒绝 $H_0$；$p$ 越小拒绝越强硬.
.3 $p$ 值衡量拒绝的强度，$\alpha$ 是事先划定的门槛.
}

\end{note}

\section{正态总体检验}
\begin{theorem}[假设检验统计量选择]

\textbf{一、均值检验（单总体）}

设 $X_1, \ldots, X_n \overset{\text{iid}}{\sim} \mathcal{N}(\mu, \sigma^2)$，检验 $H_0: \mu = \mu_0$。

\textbf{方差已知} → $Z$ 检验：
$$Z = \frac{\bar{X} - \mu_0}{\sigma/\sqrt{n}} \sim \mathcal{N}(0,1)$$

\textbf{方差未知} → $t$ 检验：
$$T = \frac{\bar{X} - \mu_0}{S/\sqrt{n}} \sim t(n-1)$$

分母用 $S$ 替代 $\sigma$，引入额外随机性，分布变胖；$n$ 越大 $t$ 越接近正态。

\textbf{二、方差检验（单总体）}

检验 $H_0: \sigma^2 = \sigma_0^2$：
$$\chi^2 = \frac{(n-1)S^2}{\sigma_0^2} \sim \chi^2(n-1)$$

\textbf{三、两总体方差比检验}

检验 $H_0: \sigma_1^2 = \sigma_2^2$：
$$F = \frac{S_1^2}{S_2^2} \sim F(n_1-1,\, n_2-1)$$

\end{theorem}
检验统计量与 CI 的 pivot 虽然长大相似但是却不一样，代入的是具体μ在代入的一刻就知道他在0，1正态分布x轴落在哪里了
$$Z = \frac{\bar{X} - \mu_0}{\sigma/\sqrt{n}}, \qquad
T = \frac{\bar{X} - \mu_0}{S/\sqrt{n}}, \qquad
\chi^2 = \frac{(n-1)S^2}{\sigma_0^2}$$
\begin{dxtips}
    \begin{itemize}
    \item 第一：置信区间是要找一个区间；假设检验那个区间是拒绝域，那个区间是接受域，都是确定的。
    \item 第二：置信区间 $\theta$ 是不知道的；假设检验 $\theta_0$ 是给我们具体的，需要代入然后观察落在哪里。
    \item 第三：他们两构造的目的其实不一样——一个是更换轴，我们才能求出两边的方程；一个就是为了标准正态化，方便查表计算。
\end{itemize}
\end{dxtips}

单侧双侧拒绝域就是总积分为 $\alpha$ 的面积是都放左边、都放右边、还是两边各一半。
\begin{itemize}
    \item \textbf{双侧}：两边各放 $\dfrac{\alpha}{2}$，查 $z_{\alpha/2}$
    \item \textbf{单侧右}：全放右边，查 $z_{\alpha}$
    \item \textbf{单侧左}：全放左边，查 $-z_{\alpha}$
\end{itemize}
$H_1$ 指哪边就往哪边放。
\begin{note}
    我们现在不知道 $\theta_0$ 是什么分布，但是样本很大，CLT 保证
$$\frac{\sqrt{n}(\hat{\theta} - \theta_0)}{\sqrt{V(\theta_0)}} \approx N(0,1)$$
然后问题就变成：这个值落在 $N(0,1)$ 的 $x$ 轴哪里，有没有超过临界值 $z_\alpha$ 或 $z_{\alpha/2}$？超过了就拒绝 $H_0$，没超过就不拒绝。
\end{note}
 \textbf{$\theta$ 是谁：}
\begin{itemize}
    \item $\hat{\theta}$：估计量（比如 $\hat{p} = \bar{X}$，或 $\bar{X}$）
    \item $\theta_0$：$H_0$ 给定的具体值（比如 $p_0 = 0.1$）
    \item $V(\theta_0)$：估计量在 $H_0$ 下的渐近方差
\end{itemize}

\textbf{为什么和 CLT 长得不一样：}

你记的 CLT 基础形式是只乘 $\sqrt{n}$：
$$\sqrt{n}(\bar{X} - \mu) \xrightarrow{d} N(0,\, \sigma^2)$$
但得到的是 $N(0, \sigma^2)$，方差不是 $1$，没法直接查表。为了查表，还要再除以 $\sigma$：
$$\frac{\sqrt{n}(\bar{X} - \mu)}{\sigma} \xrightarrow{d} N(0,1)$$
大样本检验公式里的 $\sqrt{V(\theta_0)}$ 就是这个 $\sigma$——多了这一步除法，才能落到标准正态的 $x$ 轴上查表。对应关系：
\begin{itemize}
    \item $\hat{\theta} = \bar{X}$
    \item $V(\theta_0) = \sigma^2$（渐近方差，不带 $1/n$）
\end{itemize}
分母的 $\sqrt{n}$ 已经提到分子上去了，所以 $\sqrt{V(\theta_0)}$ 对应的是 $\sigma$，不是 $\sigma/\sqrt{n}$。本质完全一样。
\begin{itemize}
    \item \textbf{用 $\theta_0$}：$H_0$ 明确给了 $\theta_0$ 的值，直接代入算方差。
    理论上更严格，因为检验本身就是假设 $H_0$ 成立。
    \item \textbf{用 $\hat{\theta}$}：$\theta_0$ 对应的方差形式复杂，或题目直接用样本估计。
    大样本下 $\hat{\theta} \xrightarrow{P} \theta_0$，两者等价。
\end{itemize}
\section{NP-UMP}
 \textbf{NP引理的核心思想总结}
\begin{note}[奈曼皮尔逊]
在简单假设 $H_0: \theta = \theta_0$ vs $H_1: \theta = \theta_1$ 下，
让 power 势最优——即拒绝域大小（第一类错误概率）确定的情况下，使第二类错误概率最小：
\[
\text{约束：} P_{\theta_0}(X \in W) = \alpha, \qquad
\text{目标：} \min\, P_{\theta_1}(X \notin W)
\iff \max\, P_{\theta_1}(X \in W)
\]
即``冤枉好人''概率不超过 $\alpha$ 的前提下，使``抓住坏人''的概率最大。

第二类错误的几何形式取决于方向：右侧检验为 $H_1$ 下 $c$ 左侧面积，左侧检验为右侧面积。
统一表述：\textbf{$H_1$ 下落在拒绝域之外的概率最小}，不绑定方向。
\end{note}
\begin{dxtips}
    在 $H_0$ 分布下，以下三个 $W$ 都能让积分 $= \alpha$：
\begin{align*}
W_1 &= \{X \geq c_1\} \quad \text{右尾，} P_{\theta_0}(X \in W_1) = \alpha \\
W_2 &= \{X \leq c_2\} \quad \text{左尾，} P_{\theta_0}(X \in W_2) = \alpha \\
W_3 &= \{\text{某个中间奇怪区域}\} \quad P_{\theta_0}(X \in W_3) = \alpha
\end{align*}
三个 $W$ 在 $H_0$ 下积分面积相同，但对应的 $x$ 取值范围完全不同。
放到 $H_1$ 分布下积分，结果各异：
\[
P_{\theta_1}(X \in W_1) \neq P_{\theta_1}(X \in W_2) \neq P_{\theta_1}(X \in W_3)
\]
因为 $H_1$ 的分布中心不同，同一个 $x$ 范围在 $H_1$ 下积出的面积也不同，
三个 $W$ 对应三个不同的势，即三个不同的第二类错误。

\textbf{约束的是 $H_0$ 下的面积，优化的是 $H_1$ 下的面积，二者独立。}
NP引理保证：用似然比 $\Lambda(x) \geq k$ 定出的 $W$，在 $H_1$ 下的面积最大（势最大，第二类错误最小）。
\end{dxtips}
\begin{note}
    似然比是一个\textbf{性价比指标}，帮你决定每个 $x$ 值值不值得放进 $W$。

放一个 $x$ 进 $W$ 的代价和收益：
\begin{itemize}
    \item 代价：消耗掉一点 $\alpha$ 的预算 $\;\to\;$ $f(x;\theta_0)$ 越大，代价越高；其中 $f(x;\theta_0)$ 就是以 $\theta_0$ 为中心的正态曲线在 $x$ 处的纵坐标
    \item 收益：增加一点势 $\;\to\;$ $f(x;\theta_1)$ 越大，收益越高；其中 $f(x;\theta_1)$ 就是以 $\theta_1$ 为中心的正态曲线在 $x$ 处的纵坐标
\end{itemize}

所以每个 $x$ 的性价比就是：
\[
\Lambda(x) = \frac{f(x;\theta_1)}{f(x;\theta_0)} = \frac{\text{收益}}{\text{代价}}
= \frac{H_1\text{ 曲线纵坐标}}{H_0\text{ 曲线纵坐标}}
\]

最优策略：按性价比从高到低把 $x$ 塞进 $W$，直到 $\alpha$ 用完，等价于：
\[
W = \{\Lambda(x) \geq k\}
\]
NP引理证明的就是：这个贪心策略给出全局最优——势最大，第二类错误最小。
\end{note}
\begin{dxtips}
   $\Lambda(x) \geq k$ 越大越好，等价于：
\begin{itemize}
    \item $f(x;\theta_0)$（$x$ 在 $H_0$ 曲线的纵坐标）越小越好
    $\;\to\;$ 放进 $W$ 消耗的 $\alpha$ 预算少，代价小
    \item $f(x;\theta_1)$（$x$ 在 $H_1$ 曲线的纵坐标）越大越好
    $\;\to\;$ 在 $H_1$ 下对 $W$ 积分越大，势 $P_{\theta_1}(X \in W)$ 越大
\end{itemize}
从而第二类错误 $= 1 - P_{\theta_1}(X \in W)$ 越小。
\end{dxtips}
\begin{note}[一致最优势检验 UMP]
\textbf{UMP = Uniformly Most Powerful Test，一致最优势检验。}

NP解决的是简单 vs 简单（$H_1$ 只有一个点），但实际问题里 $H_1$ 往往是一个范围：
\[
H_1: \theta > \theta_0 \qquad (H_1 \text{ 有无穷多个点})
\]
这时对每个具体的 $\theta_1$ 用NP都能找到一个最优检验，但不同的 $\theta_1$ 可能给出不同的拒绝域。

\textbf{UMP问的是：}有没有一个拒绝域 $W$，对 $H_1$ 范围内\textbf{所有的} $\theta_1$ 同时最优？
\[
\beta_W(\theta_1) \geq \beta_{W'}(\theta_1) \quad \text{对一切 } \theta_1 \in H_1 \text{ 成立}
\]
如果存在，就叫 \textbf{UMP检验}。

\begin{itemize}
    \item NP $\;\to\;$ 简单 vs 简单，最优
    \item UMP $\;\to\;$ 简单 vs 复合，对所有 $\theta_1$ 一致最优
\end{itemize}
UMP不是总存在的，单调似然比族（如指数族）里单侧检验一般有UMP，双侧一般没有。
\end{note}
\begin{note}[UMP 的构造]
构造UMP的核心工具是\textbf{单调似然比}。

\textbf{思路：}NP告诉我们，对每个固定的 $\theta_1$，最优拒绝域是 $\{\Lambda(x) \geq k\}$。
若 $\Lambda(x) = \dfrac{f(x;\theta_1)}{f(x;\theta_0)}$ 是某个统计量 $T(X)$ 的单调递增函数，则：
\[
\Lambda(x) \geq k \iff T(X) \geq c
\]
右边的形式 $\{T(X) \geq c\}$ 与 $\theta_1$ 的具体取值无关，对所有 $\theta_1 > \theta_0$ 给出同一个拒绝域。


\end{note}
\begin{dxtips}
选取比值大于等于 $k$ 的 $x$ 构成拒绝域 $W = \{\Lambda(x) \geq k\}$，
在 $H_0$ 曲线下积分恰好等于 $\alpha$：
\[
P_{\theta_0}(X \in W) = \int_W f(x;\theta_0)\, dx = \alpha
\]
$k$ 即为让该积分恰好等于 $\alpha$ 的门槛值。也就是这些x坐标对应的y坐标乘dx再求和
\end{dxtips}
\begin{note}[单调性的证明]
取对数后：
\[
\ln \Lambda(x) = \frac{\mu_1 - \mu_0}{\sigma^2} \cdot \bar{X} - \text{常数}
\]
这是关于 $\bar{X}$ 的线性函数，系数为 $\dfrac{\mu_1-\mu_0}{\sigma^2}$，证单调只需看系数符号——正的单增，负的单减。

虽然每个 $\theta_1$ 取值不同，但只要 $H_1: \theta > \theta_0$，$\theta_1 - \theta_0 > 0$ 是确定的，
故上式形如 $k\bar{X} - b$（$k > 0$），$\bar{X}$ 越大 $\Lambda(x)$ 越大，单增方向锁死，不随 $\theta_1$ 变。

故拒绝域 $\{\bar{X} \geq c\}$ 对所有 $\theta_1 > \theta_0$ 一致成立 $\Rightarrow$ \textbf{UMP}。
\end{note}




\begin{note}[似然比化简为拒绝域]
\[
\Lambda(x) = \exp\!\left\{\frac{\mu_1-\mu_0}{\sigma^2}\sum x_i - \text{常数}\right\} \geq k
\]
两边取 $\ln$，常数移右，设右端为 $c'$：
\[
\frac{\mu_1-\mu_0}{\sigma^2}\sum x_i \geq c'
\]
因 $\mu_1 > \mu_0$，系数 $> 0$，两边除之不等号方向不变，设右端为 $c$：
\[
\Lambda(x) \geq k \iff \bar{X} \geq c
\]
$\theta_1$ 虽出现在 $c$ 的表达式中，但 $c$ 是一个具体的数，
最终由 $P_{\theta_0}(\bar{X} \geq c) = \alpha$ 定出，$\theta_1$ 被绕开。
\end{note}
\begin{dxtips}[NP到UMP的完整逻辑]
我们想知道这次实验拒绝域应该取在哪里。

由 NP 引理，理论上最优拒绝域是让拒绝域内的 $x$ 满足似然比大于等于 $k$：
\[
W = \{\Lambda(x) \geq k\}
\]
$k$ 再小一点，满足条件的 $x$ 更多，拒绝域变大，$H_0$ 下的积分面积就超过 $\alpha$。

但此时仍不知道具体拒绝域范围在哪，因为 $\Lambda(x)$ 含有未知的 $\theta_1$。

引入 $T(X)$，利用单调性：
\[
\Lambda(x) \geq k \iff T(X) \geq c, \quad c \text{ 由 } P_{\theta_0}(T(X) \geq c) = \alpha \text{ 定出}
\]
$T(X)$ 只含数据，代入样本即可算出具体拒绝域。
\end{dxtips}
\begin{note}[从似然比到可用拒绝域]
最初只知道最优拒绝域是 $\{\Lambda(x) \geq k\}$，
但由于 $\Lambda(x) = \dfrac{f(x;\theta_1)}{f(x;\theta_0)}$ 含有未知的 $\theta_1$，
不知道 $x$ 具体取哪些值，拒绝域落在哪里。

因此需要找统计量 $T(X)$，利用单调性将最优条件翻译成可计算的形式：
\[
\Lambda(x) \geq k \iff T(X) \geq c
\]
其中 $c$ 由 $P_{\theta_0}(T(X) \geq c) = \alpha$ 定出，$\theta_1$ 被绕开，$k$ 也无需单独计算。

\textbf{完整逻辑链：}
\begin{itemize}
    \item \textbf{最优性：}由 NP 引理保证，$\{\Lambda(x) \geq k\}$ 是最优拒绝域
    \item \textbf{可计算：}$T(X)$ 只含数据，拿到样本即可算出
    \item \textbf{实际判断：}计算 $T(X)$，与 $c$ 比较，判断样本是否落入拒绝域
\end{itemize}
\end{note}
\section{卡方拟合优度检验}
\begin{note}
为什么拟合优度检验要引入 $F_0$、$F_n$ 这套新符号？核心原因只有一个：\textbf{检验的对象变了}。

之前检验的是一个数，例如 $H_0: \mu = \mu_0$，统计量是拿样本统计量和这个数比。
现在要检验的是一整个分布，分布是函数，所以假设必须写成：
\[
H_0: F = F_0, \quad H_1: F \neq F_0
\]
其中 $F_0$ 是假设的理论分布，$F_n$ 是经验分布函数（用手头数据估计出来的分布）。
逻辑链升了一个层次：原来比一个数 $(\bar{X}$ vs $\mu_0)$，现在比两条曲线 $(F_n$ vs $F_0)$。

三种方法度量 $F_n$ 与 $F_0$ 的差距，方式不同：
\begin{itemize}
  \item \textbf{卡方检验}：将曲线切成 $k$ 段逐段比频数差，有拒绝域，结论客观
  \item \textbf{K-S 检验}：取 $\sup_x|F_n(x)-F_0(x)|$ 最大垂直距离，对尾部更敏感
  \item \textbf{Q-Q 图}：比分位数是否对齐画图看直线，直观但主观
\end{itemize}

这不是新体系，逻辑框架和参数检验完全一样：构造在 $H_0$ 下分布已知的统计量，超过临界值就拒绝。
\end{note}
\begin{dxtips}[P-P图的直觉理解]

我们要比较的是：\textbf{样本自己说的概率}和\textbf{理论分布说的概率}是不是一回事。

对于同一个数值 $x_{(i)}$：
\begin{itemize}
    \item 横轴 $x$：经验概率 $F_n(x_{(i)}) = i/n$，
    这个是确定的，只和排名有关。
    比如10个人最后一名55分，他就告诉了我们后百分之10对应的分位数是10\%。
    
    \item 纵轴 $y$：理论概率 $F(x_{(i)})$，
    查表得55分的分位数应该是0.045。
\end{itemize}

配对描点为 $\left(\dfrac{i}{n},\ F(x_{(i)})\right)$，
如果两者相等就能画出一条 $y = x$ 的图像。

\end{dxtips}
\begin{note}[$F_n(x_{(i)}) = i/n$ 的含义]

\begin{itemize}
    \item $n$：样本总数，比如一共10个人，$n = 10$
    \item $i$：排序后第 $i$ 个数，比如55分排第1，$i = 1$；70分排第5，$i = 5$
\end{itemize}

所以 $F_n(x_{(i)}) = i/n$ 就是「第 $i$ 个数在 $n$ 个数里的排名百分比」。

\end{note}
\begin{note}[正态分布理论分位数公式推导]

正态分布的CDF定义为：
$$F(x) = \Phi\!\left(\frac{x - \mu}{\sigma}\right)$$

已知概率 $p$，反过来问：对应的 $x$ 是多少？这就是求反函数 $F^{-1}(p)$，解得：

$$F^{-1}\!\left(\frac{i}{n}\right) = \hat{\mu} + \hat{\sigma} \cdot \Phi^{-1}\!\left(\frac{i}{n}\right)$$

其中各部分含义如下：

\begin{itemize}
    \item $\Phi^{-1}(p)$：标准正态分布的分位数，
    即标准正态分布里有 $p$ 的概率低于这个值。
    比如 $\Phi^{-1}(0.1) = -1.28$，
    意思是标准正态里有10\%的概率低于 $-1.28$。

    \item $\hat{\sigma} \cdot \Phi^{-1}(p)$：标准正态是均值0方差1的，
    乘以 $\hat{\sigma}$ 是把尺度\textbf{还原到原始数据的尺度}。

    \item $\hat{\mu} + \hat{\sigma} \cdot \Phi^{-1}(p)$：
    再加上 $\hat{\mu}$ 是把中心\textbf{平移回原始数据的均值}。
\end{itemize}

\end{note}
\begin{note}[Q-Q图（Quantile-Quantile Plot，分位数-分位数图）的横纵坐标]

\begin{itemize}
    \item 横轴 $x$：理论分位数 $F^{-1}(i/n)$，
    即理论分布说排名第 $i/n$ 的人应该是多少。

    \item 纵轴 $y$：样本分位数 $x_{(i)}$，
    即样本里实际排名第 $i$ 的人是多少。
\end{itemize}

若样本来自该分布，两者相等，点落在对角线 $y = x$ 上。

\end{note}
\begin{definition}[Kolmogorov-Smirnov 检验]
设 $X_1, \ldots, X_n \overset{i.i.d.}{\sim} F(x)$，经验分布函数为
$F_n(x) = \frac{1}{n}\sum_{i=1}^n \mathbf{1}(X_i \leq x)$。
对假设 $H_0: F(x) = F_0(x)$（$F_0$ 完全已知且连续），定义统计量
$$D_n = \sup_{x\in\mathbb{R}} |F_n(x) - F_0(x)|$$
在 $H_0$ 下，$\sqrt{n}\,D_n \xrightarrow{d} K(\lambda)$，其中
$K(\lambda) = \sum_{k=-\infty}^{+\infty}(-1)^k e^{-2k^2\lambda^2}$。
给定显著性水平 $\alpha$，拒绝域为 $\sqrt{n}\,D_n > \lambda_\alpha$。
\end{definition}
\chapter{能力五综合应用}
\section{最小二乘与检验}
\begin{note}
\textbf{模型在说什么}

你观测到一堆点 $(x_i, Y_i)$，假设它们之间有线性关系，但不是完美的直线，每个点都有误差：
\[
Y = \beta_0 + \beta_1 x + \varepsilon
\]
$\varepsilon$ 是随机噪声，假设服从 $N(0,\sigma^2)$。$x$ 是你控制的（比如广告投入），$Y$ 是结果（比如销量），随机性全来自 $\varepsilon$。

\textbf{最终要检验的是什么}

核心假设是 $H_0: \beta_1 = 0$，翻译成人话：$x$ 对 $Y$ \textbf{根本没有线性影响}，你以为有关系，其实是噪声造成的。
\begin{itemize}
  \item 拒绝 $H_0$：$x$ 和 $Y$ 之间的线性关系是显著的，这条直线是有意义的
  \item 不能拒绝 $H_0$：你拟合的直线可能就是在拟合噪声，$\hat{\beta}_1$ 虽然不等于零但完全可能是随机波动
\end{itemize}

所以整个流程是：先假设线性 $\to$ 用 OLS 估出直线 $\to$ 检验这条直线是否真的有意义（$\beta_1 \neq 0$）$\to$ 如果有意义再去做预测和区间估计。
\end{note}
\begin{dxtips}
    回归模型就是y=kx+b，k也就是x对y对影响程度b也就是为了让某些是0的时刻有初始条件
\end{dxtips}

    \begin{note}
记忆：$\hat\beta_1=S_{xy}/S_{xx}=\widehat{\mathrm{Cov}}(x,Y)/\widehat{\mathrm{Var}}(x)$，
即斜率 $=$ 协方差 $\div$ 方差，与相关系数公式同根，仅少除 $\sigma_Y$。
截距由回归直线过中心点 $(\bar x,\bar Y)$ 反推：$\hat\beta_0=\bar Y-\hat\beta_1\bar x$。
\end{note}

\begin{dxtips}
   最小二乘就是为了残差平方和最小，也就是 $(Y - kx - b)^2$ 再求和，由于这里面有两个变量，最小值就是先对两个变量分别求偏导都是 $0$。
\end{dxtips}
\begin{note}
最小二乘的目的就是算出 $x$ 前面的系数 $\hat{\beta}_1$，$H_0$ 默认是 $0$，也就是不相关。但 $\hat{\beta}_1 \neq 0$ 不够——万一只是随机波动碰巧算出来的呢？

核心问题是：$\hat{\beta}_1$ 离零有多远，远到不像噪声造成的？

光看 $\hat{\beta}_1$ 本身没意义，单位不同你不知道大还是小，所以标准化：
\[
T = \frac{\hat{\beta}_1}{\hat{\sigma}/\sqrt{S_{xx}}} \sim t(n-2) \quad (H_0 \text{ 下})
\]
$T$ 超过临界值就拒绝 $H_0$，本质在问：\textbf{估出来的斜率，相对于数据的噪声水平，够不够大？}
\end{note}
\begin{itemize}
  \item $S_{xx} = \sum(x_i - \bar{x})^2$ 是 $x$ 的离散程度。直觉：$x$ 取值越分散，斜率估计越准——如果所有 $x$ 都挤在一起，根本看不出斜率是多少。
  \item $\hat{\sigma} = \sqrt{SS_e/(n-2)}$ 是噪声的估计，即残差的标准差——数据点离拟合直线平均有多远，越大说明数据越乱。
  \item $T$ 的直觉：分子 $\hat{\beta}_1$ 是斜率有多大，分母 $\hat{\sigma}/\sqrt{S_{xx}}$ 是考虑到数据有多乱、$x$ 有多散之后斜率估计的不确定性。$T$ 大 $=$ 斜率相对于噪声够显著 $\Rightarrow$ 拒绝 $H_0$。
\[
T = \frac{\hat{\beta}_1}{\hat{\sigma}/\sqrt{S_{xx}}} \sim t(n-2) \quad (H_0 \text{ 下})
\]
\end{itemize}
\begin{dxtips}
检验统计量 $T = \hat{\beta}_1 / (\hat{\sigma}/\sqrt{S_{xx}})$ 里需要 $\sigma$，但 $\sigma$ 未知，要从数据估计。用残差来估：
\[
\hat{\sigma}^2 = \frac{SS_e}{n-2}, \qquad SS_e = \sum(Y_i - \hat{Y}_i)^2 = S_{yy} - \frac{S_{xy}^2}{S_{xx}}
\]
直觉：$S_{yy}$ 是 $Y$ 的总变动，$S_{xy}^2/S_{xx}$ 是被直线解释掉的部分，剩下的就是残差。

所以流程是：$SS_e \to \hat{\sigma} \to T \to$ 查 $t(n-2)$ 表 $\to$ 拒绝或不拒绝 $H_0$。
\end{dxtips}
\begin{itemize}
  \item \textbf{背景}：已有 $\hat{Y} = 43.5 + 7.5x$，新来 $x^* = 3$，有两个不同的问题：
  \begin{enumerate}
    \item $x=3$ 时，所有学生的平均成绩是多少？（均值置信区间）
    \item $x=3$ 时，新来一个具体学生会考多少分？（个体预测区间）
  \end{enumerate}

  \item \textbf{均值置信区间}：估的是直线上那个点 $E(Y|x^*) = \beta_0 + \beta_1 x^*$，不确定性只来自 $\hat{\beta}_0, \hat{\beta}_1$ 的估计误差：
  \[
  \hat{Y}^* \pm t_{\alpha/2}(n-2) \cdot \hat{\sigma}\sqrt{\frac{1}{n} + \frac{(x^*-\bar{x})^2}{S_{xx}}}
  \]

  \item \textbf{个体预测区间}：估的是新个体 $Y^* = \beta_0 + \beta_1 x^* + \varepsilon^*$，不确定性来自两处：参数估计误差 $+$ 新个体自己的随机噪声 $\varepsilon^*$：
  \[
  \hat{Y}^* \pm t_{\alpha/2}(n-2) \cdot \hat{\sigma}\sqrt{1 + \frac{1}{n} + \frac{(x^*-\bar{x})^2}{S_{xx}}}
  \]
  多出来的"$1$"就是 $\varepsilon^*$ 贡献的方差，所以预测区间永远比置信区间宽。
\end{itemize}
\begin{note}
两种区间形式完全一样：
\[
\hat{Y}^* \pm t_{\alpha/2}(n-2) \cdot SE
\]
查的是同一张 $t(n-2)$ 表，区别只在 $SE$ 公式不同。
\end{note}
\section{方差分析}
\begin{note}
\textbf{为什么不直接两两 $t$ 检验？}

有 $k$ 个组，两两检验需要做 $\binom{k}{2}$ 次。每次 $\alpha = 0.05$，但整体至少犯一次错的概率：
\[
1 - (1-0.05)^{\binom{k}{2}}
\]
组数越多错误率膨胀越离谱，$k=4$ 时已经达到 $26\%$。

\textbf{ANOVA（Analysis of Variance，方差分析）}的思路是一次检验解决问题：
\[
H_0: \mu_1 = \mu_2 = \cdots = \mu_k
\]
用一个统计量 $F$ 判断，整体错误率就是设定的 $\alpha$，不会膨胀。

名字叫"方差分析"而不叫"均值分析"，是因为它的做法是把数据总变动拆成两部分来比较，用的是方差的语言（$SS_T = SS_A + SS_E$）。
\end{note}
\begin{definition}[单因子方差分析模型]
\[
y_{ij} = \mu + \mu_i + e_{ij}, \quad e_{ij} \sim N(0, \sigma^2) \text{ iid}
\]
$k$ 个组，第 $i$ 组有 $n_i$ 个观测，$N = \sum n_i$，约束 $\sum \mu_i = 0$。
\end{definition}

\begin{itemize}
  \item $\mu$：所有组共同的总体均值，"基准线"
  \item $\mu_i$：第 $i$ 组相对于基准线的偏移，"这组特别在哪"
  \item $e_{ij}$：个体随机噪声
  \item 约束 $\sum \mu_i = 0$ 是为了让分解唯一——否则 $\mu$ 加大、$\mu_i$ 减小，无数种组合都能表示同一个值
  \item $H_0: \mu_1 = \cdots = \mu_k = 0$，即所有组偏移为零，没有组间差异
\end{itemize}
\begin{note}
平方和分解 $SS_T = SS_A + SS_E$，数据的总波动拆成组间和组内两部分：
\begin{itemize}
  \item $SS_T = \sum_i\sum_j (y_{ij} - \bar{y})^2$：所有观测值离总均值的距离，衡量数据整体有多散
  \item $SS_A = \sum_i n_i(\bar{y}_i - \bar{y})^2$：每组均值离总均值的距离，衡量各组之间差异有多大，各组均值差不多则 $SS_A$ 小
  \item $SS_E = \sum_i\sum_j (y_{ij} - \bar{y}_i)^2$：每个观测值离自己组均值的距离，衡量组内随机噪声，纯粹是误差
\end{itemize}

核心逻辑：用 $F$ 统计量比较组间和组内：
\[
F = \frac{MS_A}{MS_E} = \frac{SS_A/(k-1)}{SS_E/(N-k)}
\]
\begin{itemize}
  \item $H_0$ 下：$MS_A$ 和 $MS_E$ 都是 $\sigma^2$ 的无偏估计，比值应该接近 $1$
  \item $H_1$ 下：组间差异让 $MS_A$ 偏大，$F \gg 1$ 说明不像随机波动
\end{itemize}
$F > F_\alpha(k-1, N-k)$ 则拒绝 $H_0$。
\end{note}
\begin{dxtips}
平方和分解类比柯赫伦定理：$SS_T$ 自由度 $N-1$ 被正交拆成两块：
\[
\underbrace{SS_T}_{N-1} = \underbrace{SS_A}_{k-1} + \underbrace{SS_E}_{N-k}
\]
柯赫伦定理保证 $H_0$ 下 $SS_A$ 与 $SS_E$ 相互独立，且：
\[
\frac{SS_A}{\sigma^2} \sim \chi^2(k-1), \qquad \frac{SS_E}{\sigma^2} \sim \chi^2(N-k)
\]
于是 $F$ 统计量服从 $F$ 分布直接来自 $F$ 分布的定义——两个独立卡方除以各自自由度之比：
\[
F = \frac{MS_A}{MS_E} = \frac{SS_A/(k-1)}{SS_E/(N-k)} \sim F(k-1, N-k)
\]
不是凭空规定的，是柯赫伦定理的直接推论。
\end{dxtips}
